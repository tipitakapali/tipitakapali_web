នមោ តស្ស ភគវតោ អរហតោ សម្មាសម្ពុទ្ធស្ស

លក្ខណាតោ

ពុទ្ធថោមនាគាថា

១.

ភវាភវេសុ ¶ នេកេសុ,

បូរេសិ តិំសបារមី;

ចរិយាយោច សម្ពុទ្ធោ,

បរិច្ចាគេជហំសទា.

២.

វេស្សន្តរត្តភាវម្ហិ ¶ ,

ឋិតោមទ្ទិំ បិសោអទា;

និជំកណ្ហាជិនំជាលិំ,

វិយំវង្កតបព្ពតេ.

៣.

តតោចវិត្វាតុសិតេ,

បុរេឧប្បជ្ជិបុញ្ញវា;

រូបាទិទសឋានេហិ ¶ ,

អញ្ញេទេវេអតិក្កមិ.

៤.

ចក្កវាឡសហស្សេហិ,

ទសហាគម្មសព្ពសោ;

យាចិតោទេវព្រហ្មូហិ,

ពុទ្ធភាវមនោនតោ.

៥.

វិលោកនានិសោបញ្ច ¶ ,

វិលោកេត្វាសិរិន្ធរោ;

តតោចវិត្វាមាយាយ,

ឧប្បជ្ជិសាកិយេកុលេ.

៦.

វិសាខបុណ្ណមេសេដ្ឋេ ¶ ,

វិជាយិលុម្ពិនីវនេ;

សីហនាទំបនាទេន្តោ,

សត្តបាទំគតោតទា.

៨.

វិរោចិំសុតទាកាសេ ¶ ,

តារោចន្ទទិវាករា;

រតនានិចភូមដ្ឋា,

នានាបុប្ផាចបុប្ផរេ.

៨.

សីតាយន្តិ អវីចគ្គី ¶ ,

ខុប្បិបាសាចសោហិតា;

លោកន្តរេមហាបភា,

អន្ធាបស្សន្តិតាវទេ.

៩.

អឃដ្ដិតាវាទិតានិ ¶ ,

តូរិយានិចអម្ពរេ;

មធុរេនបនាទិំសុ,

យសំសរេនតាវទេ.

១០.

ថុតិគីតានិគាយន្តា,

ទេវាចកមលាសនា;

សម្មោទេន្តាវអដ្ឋំសុ ¶ ,

ពុទ្ធង្កុរស្សសន្តិកេ.

១១.

កុមារោចារុធាតីហិ,

តោសិតារាជមន្ទិរេ;

វុទ្ធោសិសុសុខំខេមំ,

សក្យកញ្ញានមន្តរេ.

១២.

សោឡសវស្សិកោសក្យ ¶ ,

នន្ទនោបិតរាសតា;

រាជត្តេអភិសិត្តោសិ,

មោទេសិតុវនត្តយេ.

១៣.

នន្ទនានំសោវណ្ណ ¶ ,

វណ្ណសោភំយសោធរំ;

សោការេសិមហេសិត្តំ,

លោកនេត្តរសាយនំ.

១៤.

ឯកូនតិំសមេវស្សេ,

វិសាខបុណ្ណមេវរេ;

សហសេនាយឧយ្យាសិ,

យានេនុយ្យានមណ្ឌលំ.

១៥.

និមិត្តេចតុរោទិស្វា ¶ ,

បុត្តស្សចវិជាយនំ;

សុត្វាឧព្ពិគ្គចិត្តេន,

នរមិត្ថបុរេតហិំ.

១៦.

បច្ចាគម្មបុរំតម្ហា ¶ ,

បត្តោបាសាទមណ្ឌលំ;

និប្បជ្ជិទេវរាជាវ,

សយនេសោមហារហេ.

១៧.

សុន្ទរីតំបុរក្ខិត្វា ¶ ,

នច្ចគីតេនតោសយុំ;

ឯវំបិសោនរមិត្វា,

អាហុនិក្ខមមានសោ.

១៨.

បថមំរាហុលំបុត្តំ,

បស្សិត្វានិក្ខមាមហំ;

ឥតិចិន្តិយមានេសោ,

អគាជាយានិវេសនំ;

១៩.

ថបេត្វាបាទទុម្មារេ ¶ ,

គីវមន្តោបវេសយំ;

ឱលោកយំសបស្សិត្ថ,

និប្បន្នំតំសមាតរំ.

២០.

គណ្ហេយ្យំតំអបនេត្វា ¶ ,

ពាហុំចេទេវិយាអហំ;

នទទេនិក្ខមោកាសំ,

បពុជ្ឈិត្វាយសោធរា.

២១.

ពុទ្ធបត្តោវបស្សេយ្យំ,

រាហុលំឥតិចិន្តិយ;

កណ្ឌកំអារុហិត្វាន,

តម្ហាឆន្នេននិក្ខមិ.

២២.

ទ្វារាសយាវិវរិំសុ ¶ ,

ទេវតាទ្វារមណ្ឌលំ;

រតនុក្កាសហស្សានិ,

ជាលយិំសុមរូតហិំ.

២៣.

មានិក្ខមស្សុសិទ្ធត្ថ,

ឥតោត្វំសត្តមេទិនេ;

ភវិស្សសេចក្កវត្តី ¶ ,

ឥតិនីវារិបាបិមា.

២៤.

តគ្ឃជានាមិតំមារ,

ឥធមាតិដ្ឋបក្កម;

ឥតិមារំ បលាបេត្វា,

វិស្សដ្ឋោសោអភិក្កមិ.

២៥.

បត្វានោមនទីតីរំ ¶ ,

បុនាគមេសិតុរគំ;

ឆន្នញ្ចនគរំ ពិម្ពា;

ទេវិលាបដិវេទិតុំ.

២៦.

ពន្ធិតំយសោធរាយ,

មោឡិំសុគន្ធវាសិតំ;

ឆេត្វាខគ្គេនតិណ្ហេន,

ឧក្ខិបិត្ថតមម្ពរេ.

២៧.

អដ្ឋសេដ្ឋបរិក្ខារេ ¶ ,

ធារេត្វាព្រហ្មុនាភតេ;

ហិត្វារាជារហំទុស្សំ,

សមណត្ថមុបាគមិ.

២៨.

ករោន្តោទុក្ករំកម្មំ ¶ ,

ឆព្ពស្សានិបរក្កមោ;

ឱឡារិកន្នបានេហិ;

ទេហំយាបេសិឋីតិយា.

២៩.

វិសាខបុណ្ណមេហេម,

បាតិំបាយសបូរិតំ;

សុជាតាយាភតំគយ្ហ,

នេរញ្ចរមុបាគមិ.

៣០.

នេរញ្ចរាយតីរម្ហិ ¶ ,

សុត្វាបាយាសមុត្តមំ;

បដិយត្ថវរមគ្គេន,

ពោធិមណ្ឌមុបាគមិ.

៣១.

ជយាសនេបល្លង្កម្ហិ ¶ ,

ឋិតោធិដ្ឋេសិធីតិមា;

នវីរិយំជហិស្សាមិ,

អបត្តោឥតិពុទ្ធតំ.

៣២.

ទេវិន្ទោធមយំសង្ខំ ¶ ,

សេតច្ឆត្តំបិតាមហោ;

ធារយំវាទេន្តោវីណំ,

បញ្ចសីខោឋិតោតហិំ.

៣៣.

ថុតិគីតានិគាយន្តី,

អដ្ឋាសុំទេវសុន្ទរី;

គហេត្វាហេមមញ្ជូសា,

សុរបុប្ផេហិបូរិតា.

៣៤.

អនត្ថមេន្តេសូរិយេ ¶ ,

មារសេនំបលាបយី;

អរុណុគ្គមនេកាលេ,

បត្តោសិពុទ្ធតំវរំ.

៣៥.

បដ្ឋានំសម្មសន្តស្ស,

នាថស្សទេហតោសុភា;

ឆព្ពណ្ណភានិច្ឆធន្តិ ¶ ,

វិធាវន្តិ តហិំ តហិំ.

៣៦.

នីលានីលាបិតាបីតា,

រត្តារត្តាចនិច្ឆរេ;

សេតាសេតាចមញ្ចិដ្ឋា,

តម្ហាតម្ហាបភស្សរា.

៣៧.

ពុទ្ធបត្តោបិពោធេយ្យ ¶ ,

ធម្មគម្ភីរតំជិនោ;

និសាមេត្វានទេសេតុំ,

និរុស្សាហមុបាគតោ.

៣៨.

សហម្បតិតទាព្រហ្មា ¶ ,

ទេសនាយាភិយាចតិ;

តេនសោបថមំសេដ្ឋំ,

ធម្មចក្កមទេសយិ.

៣៩.

ឯកវស្សោវសម្ពុទ្ធោ ¶ ,

វរេផគ្គុណបុណ្ណមេ;

សក្យានំនគរំគញ្ឆី,

សាវកេហិបុរក្ខតោ.

៤០.

សមាគមម្ហិញាតីនំ,

វស្សាបេសិសុសីតលំ;

មហាមេឃញ្ចទេសេសិ ¶ ,

វេស្សន្តរស្សជាតកំ.

៤១.

គន្ត្វាសាវកយុគេន,

យសោធរាយមន្ទិរំ;

តំតោសេត្វានទេសេសិ,

ចន្ទកិន្នរជាតកំ.

៤២.

វសិត្វាសត្តមេវស្សេ ¶ ,

តាវតិំសេ សិលាសនេ;

អភិធម្មេន តោសេសិ,

ទេវព្រហ្មគណំពហុំ.

៤៣.

បត្ថរំញាណជាលេន,

លោកំលោកេសិសព្ពទា;

ពោធេតុំពោធិយាភាយ;

ពោធេយ្យជនបង្កជំ.

៤៤.

ពោធនេយ្យំជនំទិស្វា ¶ ,

ចក្កវាឡបរំបរា;

គន្ត្វាបិវិពោធេសិ,

ពុទ្ធសេដ្ឋោសុពោធិមា.

៤៥.

ពកាទិកេចព្រហ្មានោ,

យក្ខេចាឡវកាទិកេ;

បុរិន្ទទាទិកេទេវេ,

សូចិលោមាទិរក្ខសេ.

៤៦.

ជាណុសោណាទិកេនេកេ ¶ ,

ព្រាហ្មណេចាវិកក្ខឡេ;

ចូឡោទរាទិកេនាគេ,

ធនបាលាទិកេគជេ.

៤៧.

ពុទ្ធាមយំតិមញ្ញន្តេ,

តិត្ថិយេនេកលទ្ធិកេ;

ទមេសិទមតំសេដ្ឋោ,

បរវាទប្បមទ្ទនោ.

៤៨.

អនុព្យញ្ជនសម្បន្នោ ¶ ,

ឯសពាត្តិំសលក្ខណោ;

ព្យាមប្បភាបរិក្ខិត្តោ,

កេតុមាលាសិលង្ខតោ.

៤៩.

ចតុរាសីតិសហស្ស ¶ ,

ធម្មក្ខន្ធេសទេសយិ;

បញ្ចតាថីសវស្សានិ,

តិលោកស្សហិតាយវេ.

៥០.

ពុទ្ធកិច្ចំករោន្តោសោ,

វហំលោកហិតំវរំ;

មោចយំពន្ធនាសត្តេ ¶ ,

កាលំខេបេតិសព្ពទា.

៥១.

យហិំ អាសីតិកោនាថោ,

និព្ពុតោវរមញ្ជកេ;

មហាសាលវនេរម្មេ,

មល្លានំនិគមេតហិំ.

៥២.

អគ្គិក្ខន្ធោវលោកស្មិំ ¶ ,

ជាលេត្វានសសាវកោ;

និព្ពាយិត្ថសង្ខតានំ,

បកាសេន្តោអនិច្ចតំតិ.

‘‘នមោតេភគវានាថ ¶ ,

នមោតេជយតំជយ;

នមោតេសិរិសម្បន្ន,

នមោតេលោកនន្ទន.’’

ពុទ្ធវន្ទនា

១.

ទេវលោកា ចវិត្វាន ¶ ,

មហាមាយាយ កុច្ឆិយំ;

ឧប្បជ្ជិ គុរុវារម្ហិ,

វន្ទេតំ សក្យបុង្គវំ.

២.

ទសមា សច្ចយេ នេសោ ¶ ,

វិជាយិ មាតុកុច្ឆិតោ;

សុក្កវារេ លុម្ពិនិយំ,

វន្ទេ តំ លោកបូជិតំ.

៣.

ចក្កវត្តិសិរិំ ហិត្វា,

មហាសិនិក្ខមំ សុធី;

និក្ខមី ចន្ទវារម្ហិ,

នមេ តំ មុនិកុញ្ជរំ.

៤.

អស្សត្ថពោធិមូលម្ហិ ¶ ,

បល្លង្កេ អបរាជិតេ;

បត្តោ សព្ពញ្ញុតំ នាថោ,

នមេ តំ ពុធវាសរេ.

៥.

បញ្ចវគ្គិយមុខានំ ¶ ,

ទេវានំ មិគទាវនេ;

សោរិវារេ ធម្មចក្កំ,

វត្តេសិ តំ ជិនំ នមេ.

៦.

សព្ពសង្ខតធម្មានំ,

បកាសិយអនិច្ចតំ;

និព្ពុតំអង្គវារម្ហិ,

នមាធិអមតន្ទទំ.

៧.

សូរិយវារេ សោណ្ណាយ ¶ ,

ទោណិយា ជិនវិគ្គហោ;

អគ្គិនុជ្ជលិតោ តស្ស,

នមេ ធាតុសរីរកំ.

ឧណ្ណាលោមិកនាថ វន្ទនា

ឧណ្ណាលោធិកនាថស្ស ¶ ,

ឧណ្ណាយ ភមុកន្តរេ;

វជិរាវិយ សោភន្តិ,

និក្ខន្តិ យោ សុបណ្ឌរា.

វជិរគ្ឃនកាយស្ស ¶ ,

នាថស្ស ទេហតោ សុភា;

វជិរេយ្យា និច្ឆរន្តិ;

វិជ្ជូវ គគណន្តរេ.

រិនិន្ទមន្ទហាសេន,

ទាឋាយ និច្ឆរា បភា;

វិប្ផូរិសា ទិសាសព្ពា,

វជិរាយន្តិ បណ្ឌរា.
