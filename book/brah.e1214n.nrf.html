𑀓𑀘𑁆𑀘𑀸𑀬𑀦

𑀢𑀼𑀤𑀸𑀤𑀬𑁄 𑀅𑀯𑀼𑀤𑁆𑀥𑀺𑀓𑀸

𑀭𑀼𑀥𑀸𑀤𑀬𑁄

𑀲𑀸𑀘𑀭𑀺𑀬𑀸𑀦𑀼𑀲𑀺𑀝𑁆𑀞𑀸

𑀓𑀘𑁆𑀘𑀸𑀬𑀦 ¶ 𑀥𑀸𑀢𑀼 𑀫𑀜𑁆𑀚𑀽𑀲𑀸

𑀦𑀫𑁄 𑀢𑀲𑁆𑀲 𑀪𑀕𑀯𑀢𑁄 𑀅𑀭𑀳𑀢𑁄 𑀲𑀫𑁆𑀫𑀸𑀲𑀫𑁆𑀩𑀼𑀤𑁆𑀥𑀲𑁆𑀲

𑀦𑀺𑀭𑀼𑀢𑁆𑀢𑀺 𑀦𑀺𑀓𑀭𑀸’𑀧𑀸𑀭-𑀧𑀸𑀭𑀯𑀸𑀭’𑀦𑁆𑀢𑀕𑀁 𑀫𑀼𑀦𑀺𑀁,

𑀯𑀦𑁆𑀤𑀺𑀢𑁆𑀯𑀸 𑀥𑀸𑀢𑀼𑀫𑀜𑁆𑀚𑀽𑀲𑀁-𑀩𑁆𑀭𑀽𑀫𑀺 𑀧𑀸𑀯𑀘𑀦𑀜𑁆𑀚𑀲𑀁.

𑀲𑁄𑀕𑀢𑀸𑀕𑀫 𑀫𑀸’𑀕𑀫𑁆𑀫-𑀢𑀁 𑀢𑀁𑀯𑁆𑀬𑀸𑀓𑀭𑀡𑀸𑀦𑀺 𑀘,

𑀧𑀸𑀞𑁂 𑀘𑀸’𑀧𑀞𑀺𑀢𑀸𑀧𑁂’𑀢𑁆𑀣 𑀥𑀸𑀢𑁆𑀯𑀢𑁆𑀣𑀸 𑀘 𑀧𑀯𑀼𑀘𑁆𑀘𑀭𑁂.

𑀙𑀦𑁆𑀤’𑀳𑀸𑀦𑀺𑀢𑁆𑀣𑀫𑁄’𑀓𑀸𑀭𑀁-𑀥𑀸𑀢𑁆𑀯𑀦𑁆𑀢𑀸𑀦𑀁 ¶ 𑀲𑀺𑀬𑀸𑀓𑁆𑀯 𑀘𑀺, 𑀬𑀽𑀦𑀁 𑀤𑀻𑀖𑁄 𑀘 𑀥𑀸𑀢𑀼𑀫𑁆𑀳𑀸-𑀧𑀼𑀩𑁆𑀩𑀫’𑀢𑁆𑀣𑀧𑀤𑀁 𑀅𑀧𑀺.

𑁧.

𑀪𑀽 𑀲𑀢𑁆𑀢𑀸𑀬𑀁 𑀧𑀘 𑀧𑀸𑀓𑁂 𑀕𑀫𑀼𑀲𑀧𑁆𑀧 𑀕𑀢𑀺𑀫𑁆𑀳𑀺 (𑀘);

𑀲𑀺𑀮𑁄𑀓 (𑀥𑀸𑀢𑀼) 𑀲𑀗𑁆𑀖𑀸𑀢𑁂 𑀲𑀓𑀺 𑀲𑀗𑁆𑀓𑀸𑀬 (𑀯𑀢𑁆𑀢𑀢𑁂;).

𑁨.

(𑀅𑀣𑁄) 𑀓𑀼𑀓-𑀯𑀓𑀸’𑀤𑀸𑀦𑁂 𑀓𑁂 𑀲𑀤𑁆𑀤𑁂 𑀅𑀓𑀺 𑀮𑀓𑁆𑀔𑀡𑁂;

𑀓𑀼 𑀲𑀤𑁆𑀤𑁂 𑀓𑀼𑀘𑁆𑀙𑀺𑀢𑁂 𑀝𑀗𑁆𑀓 𑀥𑀸𑀭𑀡𑁂 𑀫𑀓𑀺 𑀫𑀡𑁆𑀟𑀦𑁂.

𑁩.

𑀯𑀓𑀺 𑀓𑁄𑀝𑀺𑀮𑁆𑀮𑀬𑀸𑀢𑁆𑀭𑀸𑀲𑀼 𑀲𑀓𑁆𑀓-𑀝𑀻𑀓𑀤𑁆𑀯𑀬𑀁 𑀕𑀢𑁂;

𑀓𑀓𑀺 𑀮𑁄𑀮𑀢𑁆𑀢𑀦𑁂 𑀬𑀸𑀢𑁂 𑀢𑀓𑀻 (𑀇𑀥) 𑀕𑀢𑀸𑀤𑀺𑀲𑀼.

𑁪. ¶ 𑀯𑀯, 𑀮𑁄𑀓𑀦𑀯𑀺𑀢𑁆𑀢𑀺𑀲𑀼 𑀘𑀓𑁆𑀔𑀯𑀼𑀢𑀺𑀫𑁆𑀳𑀺 (𑀢𑀼) 𑀭𑀼𑀓𑁆𑀔 (𑀘) 𑀔𑁂 𑀣𑀺𑀭𑀳𑀺𑀁𑀲𑀔𑀡𑁂 𑀦𑀺𑀬, 𑀫𑁄’𑀧𑀦𑀬𑀺𑀝𑁆𑀞𑀺 𑀯𑀢𑀸𑀤𑀺𑀲 𑀫𑀼𑀡𑁆𑀟𑀺𑀲𑀼 𑀤𑀺𑀓𑁆𑀔 (’𑀣) 𑀓𑀓𑁆𑀔-𑀓𑀔𑀸 𑀳𑀲𑀦𑁂 𑀢𑀼𑀭, 𑀳𑀺𑀁𑀲𑀦𑀯𑀼𑀤𑁆𑀥𑀺𑀕𑀢𑀻𑀲𑀼 (𑀳𑀺) 𑀤𑀓𑁆𑀔’𑀤𑀦𑀫𑁆𑀳𑀺 (𑀢𑀼) 𑀚𑀓𑁆𑀔 (𑀘) 𑀪𑀓𑁆𑀔 (𑀫𑀢𑀸) 𑀅𑀦, 𑀚𑀸𑀮𑀤𑀼𑀔𑁂𑀲𑀼 (𑀢𑀼) 𑀤𑀺𑀓𑁆𑀔 (𑀘) 𑀤𑀼𑀓𑁆𑀔 (𑀘) 𑀇𑀓𑁆𑀔 𑀤𑀺𑀲’𑀗𑁆𑀓 𑀦 𑀓𑁄’𑀔 𑀲𑀼𑀲𑁂.

𑁫. [𑀅] 𑀦𑀺𑀓𑁆𑀔 𑀘𑀼𑀫𑁆𑀩𑀦𑁂’(𑀧𑀺) 𑀲𑀺𑀓𑁆𑀔 𑀯𑀺𑀚𑁆𑀚𑀼’𑀧𑀸𑀤𑀼’ 𑀧𑀸𑀲𑀸𑀦𑀫𑁆𑀳𑀺 𑀭𑀓𑁆𑀔 𑀕𑀼𑀢𑁆𑀢𑀺𑀯𑀸𑀭𑀡𑁂 (𑀧𑀺) 𑀉𑀜𑁆𑀙𑀦𑁂 (𑀲𑀺𑀬𑀸’𑀧𑀺) 𑀪𑀺𑀓𑁆𑀔 𑀬𑀸𑀯𑀮𑀤𑁆𑀥𑁆𑀬’𑀮𑀤𑁆𑀥𑀺𑀲𑀽 (𑀧𑀺) 𑀯𑀓𑁆𑀔 𑀭𑁄𑀲𑀲𑀁𑀳𑀢𑁂𑀲𑀼 𑀫𑁄𑀓𑁆𑀔 𑀫𑀼𑀢𑁆𑀢𑀺𑀬𑀁 𑀘𑀚𑁂 (𑀧𑀺) 𑀘𑀺𑀓𑁆𑀔 𑀯𑀸𑀘𑀩𑁄𑀥𑀦𑁂𑀲𑀼.

[𑀩] ¶ 𑀦𑀔 𑀫𑀔 𑀭𑀔 𑀦𑀗𑁆𑀔𑀸𑀫𑀗𑁆𑀔𑀭𑀓𑁆𑀔𑀻’𑀔𑀻𑀮𑀗𑁆𑀔𑀸 𑀮𑀔 𑀯𑀔 𑀇𑀔 𑀇𑀗𑁆𑀔𑀸 𑀉𑀗𑁆𑀔 𑀯𑀗𑁆𑀔𑀽’𑀔 𑀕𑀢𑁆𑀬𑀁 𑀯𑀔𑀺 𑀫𑀔𑀺 𑀓𑀔𑀺 𑀓𑀗𑁆𑀔𑁂 𑀔𑀻 𑀔𑀬𑁂 𑀉𑀓𑁆𑀔 𑀲𑁂𑀓𑁂 𑀔𑀼 𑀔𑀼𑀢𑀥𑀦𑀺𑀲𑀼 (𑀯𑀼𑀢𑁆𑀢𑁄) 𑀔𑁂(’𑀣) 𑀔𑀸𑀤𑁂 𑀲𑀼𑀧𑁂 (𑀘.)

𑁬.

𑀅𑀕𑁆𑀕𑁄 (𑀢𑀼) 𑀕𑀢𑀺𑀓𑁄𑀝𑀺𑀮𑁆𑀮𑁂 𑀮𑀕 𑀲𑀗𑁆𑀕𑁂 𑀫𑀕𑁂’𑀲𑀦𑁂;

𑀅𑀕𑀻 𑀇𑀕𑀻 𑀭𑀺𑀕𑀻 𑀮𑀺𑀕𑀻 𑀯𑀕𑀻 𑀕𑀢𑁆𑀬’𑀢𑁆𑀣𑀥𑀸𑀢𑀯𑁄.

𑁭.

𑀲𑀺𑀮𑀸𑀖 𑀓𑀢𑁆𑀣𑀦𑁂 𑀚𑀕𑁆𑀖 𑀳𑀲𑀦𑁂 𑀅𑀕𑁆𑀖 𑀅𑀕𑁆𑀖𑀦𑁂;

𑀲𑀺𑀖𑀻 𑀆𑀖𑀸𑀬𑀦𑁂 (𑀳𑁄𑀢𑀺) 𑀮𑀖𑀺 𑀲𑁄𑀲𑀕𑀢𑀻𑀲𑀼 (𑀘;).

𑁮.

𑀯𑀘 𑀩𑁆𑀬𑀢𑁆𑀢𑀯𑀘𑁂 𑀬𑀸𑀘 𑀬𑀸𑀘𑀦𑁂 𑀭𑀼𑀘 𑀤𑀺𑀢𑁆𑀢𑀺𑀬𑀁;

𑀲𑀼𑀘 𑀲𑁄𑀓𑁂 𑀓𑀼𑀘 𑀲𑀤𑁆𑀤𑁂 (𑀅𑀣𑁄) 𑀯𑀺𑀘 𑀯𑀺𑀯𑁂𑀘𑀦𑁂.

𑁯. ¶

𑀅𑀜𑁆𑀘 𑀧𑀽𑀚𑀸𑀕𑀢𑁂 𑀯𑀜𑁆𑀘 𑀕𑀫𑀦𑁂 𑀓𑀺𑀜𑁆𑀘𑀸’𑀯𑀫𑀤𑁆𑀤𑀦𑁂;

𑀮𑀼𑀜𑁆𑀘𑀸’𑀧𑀦𑀬𑀦𑁂 𑀦𑀘𑁆𑀘 𑀦𑀘𑁆𑀘𑀦𑁂 𑀫𑀘 𑀭𑁄𑀘𑀦𑁂.

𑁧𑁦.

𑀅𑀘𑁆𑀘𑀸’𑀘𑁆𑀘𑀦𑁂 𑀘𑀼 𑀯𑀘𑀦𑁂 𑀲𑀘𑁄 (𑀢𑀼) 𑀲𑀫𑀯𑀸𑀬𑀦𑁂;

𑀧𑀘 𑀬𑀸𑀢𑁂 𑀓𑀘𑀺-𑀯𑀘𑁆𑀘 𑀤𑀺𑀢𑁆𑀢𑀺𑀬𑀁 𑀫𑀘𑀺 𑀥𑀸𑀭𑀡𑁂.

𑁧𑁧.

𑀧𑀼𑀘𑁆𑀙 𑀲𑀫𑁆𑀧𑀼𑀘𑁆𑀙𑀦𑁂 𑀫𑀼𑀘𑁆𑀙 𑀫𑁄𑀳𑀲𑁆𑀫𑀺𑀁 𑀮𑀜𑁆𑀙 𑀮𑀓𑁆𑀔𑀡𑁂;

𑀅𑀜𑁆𑀙𑀸’𑀬𑀸𑀫𑁂 (𑀪𑀯𑁂) 𑀧𑀼𑀜𑁆𑀙 𑀧𑀼𑀜𑁆𑀙𑀦𑁂 𑀉𑀜𑁆𑀙 𑀉𑀜𑁆𑀙𑀦𑁂.

𑁧𑁨.

𑀢𑀘𑁆𑀙𑁄 𑀢𑀦𑀼𑀓𑀺𑀭𑀬𑁂 𑀧𑀺𑀜𑁆𑀙 𑀧𑀺𑀜𑁆𑀙𑀦𑁂 𑀭𑀸𑀚 𑀤𑀺𑀢𑁆𑀢𑀺𑀬𑀁;

𑀯𑀚𑀸’𑀚𑀕𑀫𑀦𑁂 𑀭𑀜𑁆𑀚 𑀭𑀸𑀕𑁂 𑀪𑀜𑁆𑀚𑀸’𑀯𑀫𑀤𑁆𑀤𑀦𑁂.

𑁧𑁩.

𑀅𑀜𑁆𑀚𑀼 𑀩𑁆𑀬𑀢𑁆𑀢𑀺𑀕𑀢𑀻𑀓𑀦𑁆𑀢𑀺 𑀫𑀓𑁆𑀔𑀡𑁂𑀲𑁆𑀯𑁂’𑀚 𑀓𑀫𑁆𑀧𑀦𑁂;

𑀪𑀚 𑀲𑀁𑀲𑁂𑀯𑀦𑁂 𑀲𑀜𑁆𑀚 𑀲𑀗𑁆𑀕𑁂 (𑀢𑀼) 𑀇𑀜𑁆𑀚 𑀓𑀫𑁆𑀧𑀦𑁂.

𑁧𑁪. ¶

𑀬𑀚 𑀤𑁂𑀯𑀘𑁆𑀘𑀦𑁂 𑀤𑀸𑀦𑀲𑀗𑁆𑀕𑀢𑀻𑀓𑀭𑀡𑁂𑀲𑀼 (𑀘);

𑀢𑀺𑀚𑀓𑁆𑀔𑀫𑀦𑀺𑀲𑀸𑀦𑁂𑀲𑀼 𑀤𑀸𑀦𑁂(’𑀧𑀺) 𑀘𑀚 𑀳𑀸𑀦𑀺𑀬𑀁.

𑁧𑁫.

𑀲𑀚𑀸’𑀮𑀺𑀗𑁆𑀕𑀦 𑀯𑀺𑀲𑁆𑀲𑀚𑁆𑀚 𑀦𑀺𑀫𑁆𑀫𑀸𑀡𑁂 𑀫𑀼𑀚𑁆𑀚 𑀫𑀼𑀚𑁆𑀚𑀦𑁂;

𑀫𑀚𑁆𑀚 𑀲𑀁𑀲𑀼𑀤𑁆𑀥𑀺𑀬𑀁 𑀮𑀚𑁆𑀚 𑀮𑀚𑁆𑀚𑀦𑁂 𑀢𑀚𑁆𑀚 𑀢𑀚𑁆𑀚𑀦𑁂.

𑁧𑁬.

𑀅𑀚𑁆𑀚-𑀲𑀚𑁆𑀚𑀸’𑀚𑁆𑀚𑀦𑁂 𑀲𑀚𑁆𑀚 𑀦𑀺𑀫𑁆𑀫𑀸𑀡𑁂 𑀕𑀚𑁆𑀚 𑀲𑀤𑁆𑀤𑀦𑁂;

𑀕𑀼𑀚-𑀓𑀼𑀚 𑀤𑁆𑀯𑀬𑀁 𑀲𑀤𑁆𑀤𑁂 𑀅𑀔𑁆𑀬𑀢𑁆𑀢𑁂 𑀔𑀚𑁆𑀚 𑀪𑀓𑁆𑀔𑀡𑁂.

𑁧𑁭.

𑀪𑀚𑁆𑀚 𑀧𑀸𑀓𑁂 𑀯𑀺𑀚𑀺 𑀪𑀬𑀘𑀮𑀦𑁂 𑀯𑀻𑀚 𑀯𑀻𑀚𑀦𑁂;

𑀔𑀚𑀻 𑀕𑀫𑀦𑀯𑁂𑀓𑀮𑁆𑀮𑁂 𑀚𑀻 𑀚𑀬𑁂 𑀚𑀼 𑀚𑀯𑁂 (𑀲𑀺𑀬𑀸;).

𑁧𑁮.

𑀛𑁂 𑀘𑀺𑀦𑁆𑀢𑀸𑀬𑀼𑀚𑁆𑀛 𑀉𑀲𑁆𑀲𑀕𑁆𑀕𑁂 𑀕𑀫𑀦𑁂 𑀅𑀝-𑀧𑀝 𑀤𑁆𑀯𑀬𑀁;

𑀦𑀝 𑀦𑀘𑁆𑀘𑁂 𑀭𑀝 𑀧𑀭𑀺𑀪𑀸𑀲𑀦𑁂 𑀯𑀝 𑀯𑁂𑀞𑀦𑁂.

𑁧𑁯. ¶

𑀯𑀝𑁆𑀝 𑀆𑀯𑀢𑁆𑀢𑀦𑁂 𑀯𑀡𑁆𑀝 𑀯𑀡𑁆𑀝𑀢𑁆𑀣𑁂 𑀓𑀝 𑀫𑀤𑁆𑀤𑀦𑁂;

𑀨𑀼𑀝𑁄 𑀯𑀺𑀲𑀭𑀡𑀸𑀤𑀻𑀲𑀼 𑀓𑀝 𑀲𑀁𑀯𑀭𑀡𑁂 𑀕𑀢𑁂.

𑁨𑁦.

𑀖𑀼𑀝 𑀖𑁄𑀲𑁂 𑀧𑀢𑀺𑀖𑀸𑀢𑁂 𑀯𑀺𑀝’𑀓𑁆𑀓𑁄𑀲𑁂 (𑀘) 𑀧𑁂𑀲𑀦𑁂;

𑀪𑀝 𑀪𑀢𑁆𑀬𑀁 𑀓𑀼𑀝-𑀓𑁄𑀝𑁆𑀝𑀘𑁆𑀙𑁂𑀤𑀦𑁂 𑀮𑀼𑀝 𑀮𑁄𑀝𑀦𑁂.

𑁨𑁧.

𑁨𑁨.

𑀧𑀞 𑀩𑁆𑀬𑀢𑁆𑀢𑀯𑀘𑁂 𑀳𑁂𑀞 𑀩𑀸𑀥𑀸𑀬𑀁 𑀯𑁂𑀞 𑀯𑁂𑀞𑀦𑁂;

𑀲𑀼𑀞𑀻-𑀓𑀼𑀞𑀻 𑀤𑁆𑀯𑀬𑀁 𑀲𑁄𑀲𑁂 𑀧𑀻𑀞 𑀳𑀺𑀁𑀲𑀦𑀥𑀸𑀭𑀡𑁂.

𑁨𑁩.

𑀓𑀞 𑀲𑁄𑀲𑀦𑀧𑀸𑀓𑁂𑀲𑀼 𑀯𑀞 𑀣𑀼𑀮𑀢𑁆𑀢𑀦𑁂 (𑀪𑀯𑁂);

𑁨𑁪. ¶

𑀫𑀡𑁆𑀟 𑀯𑀺𑀪𑀽𑀲𑀦𑁂 𑀘𑀡𑁆𑀟 𑀘𑀡𑁆𑀟𑀺𑀓𑁆𑀓𑁂 𑀪𑀟𑀺 𑀪𑀗𑁆𑀟𑀦𑁂.

𑁨𑁫.

𑀧𑀟𑀺 𑀉𑀧𑁆𑀧𑀡𑁆𑀟𑀦𑁂 𑀮𑀺𑀗𑁆𑀕𑀯𑁂𑀓𑀮𑁆𑀮𑁂 𑀫𑀼𑀟𑀺 𑀔𑀡𑁆𑀟𑀦𑁂,

𑀕𑀟𑀺 𑀯𑀢𑁆𑀢𑁂’𑀓𑀤𑁂𑀲𑀫𑁆𑀳𑀺 𑀕𑀟𑀺 𑀲𑀦𑁆𑀦𑀺𑀯𑀬𑁂(’𑀧𑀺𑀘;);

𑁨𑁬.

𑀭𑀟𑀺-𑀏𑀭𑀟𑀺 𑀳𑀺𑀁𑀲𑀸𑀬𑀁 𑀧𑀺𑀟𑀺 𑀲𑀗𑁆𑀖𑀸𑀢𑀆𑀤𑀺𑀲𑀼,

𑀓𑀼𑀟𑀺 𑀤𑀸𑀳𑁂 𑀧𑀟𑀺 𑀕𑀢𑁂 𑀳𑀺𑀟𑀺 𑀆𑀳𑀺𑀡𑁆𑀟𑀦𑁂 (𑀲𑀺𑀬𑀸;);

𑁨𑁭.

𑀓𑀭𑀡𑁆𑀟 𑀪𑀸𑀚𑀦’𑀢𑁆𑀣𑀫𑁆𑀳𑀺 (𑀅𑀣𑁄) 𑀮𑀟𑀺 𑀚𑀺𑀕𑀼𑀘𑁆𑀙𑀦𑁂,

(𑀯𑀢𑁆𑀢𑀢𑁂) 𑀫𑁂𑀟𑀺𑀓𑁄𑀝𑀺𑀮𑁆𑀮𑁂 𑀲𑀟𑀺 𑀕𑀼𑀫𑁆𑀩𑀢𑁆𑀣𑀫𑀻𑀭𑀡𑁂;

𑁨𑁮.

(𑀅𑀣𑁄’𑀧𑀺) 𑀅𑀟𑀺 𑀅𑀡𑁆𑀟𑀢𑁆𑀣𑁂 (𑀤𑀺𑀲𑁆𑀲𑀢𑁂) 𑀢𑀼𑀟𑀺 𑀢𑁄𑀟𑀦𑁂,

𑀯𑀟𑁆𑀠 𑀲𑀁𑀯𑀟𑁆𑀠𑀦𑁂 𑀓𑀟𑁆𑀠 𑀓𑀟𑁆𑀠𑀡𑁂 𑀪𑀡 𑀪𑀸𑀲𑀦𑁂;

𑁨𑁯. ¶

𑀲𑁄𑀡 𑀯𑀡𑁆𑀡𑁂 𑀕𑀼𑀡’𑀪𑁆𑀬𑀸𑀲𑁂 𑀇𑀡-𑀨𑁂𑀡 𑀤𑁆𑀯𑀬𑀁 𑀕𑀢𑁂,

𑀧𑀡 𑀯𑁄𑀳𑀸𑀭𑀣𑁄𑀫𑁂𑀲𑀼 (𑀯𑀢𑁆𑀢𑀢𑁂) 𑀓𑀡 𑀫𑀺𑀮𑀦𑁂;

𑁩𑁦.

𑀅𑀡-𑀭𑀡-𑀓𑀡-𑀫𑀼𑀡-𑀓𑁆𑀯𑀡-𑀓𑀼𑀡 𑀲𑀤𑁆𑀤𑁂,

𑀬𑀢 𑀧𑀢𑀺𑀬𑀢𑀦𑁂 𑀚𑀼𑀢 𑀤𑀺𑀢𑁆𑀢𑀺𑀫𑁆𑀳𑀺;

𑀅𑀢-𑀧𑀢 𑀕𑀫𑀦𑁂 𑀘𑀺𑀢 𑀲𑀜𑁆𑀜𑀸𑀡𑁂,

𑀓𑀺𑀢 𑀯𑀸𑀲𑀸’𑀤𑁄 𑀯𑀢𑀼 𑀯𑀢𑁆𑀢𑀼𑀫𑁆𑀳𑀺.

𑁩𑁧.

(𑀪𑀯𑁂) 𑀓𑀢𑁆𑀣 𑀲𑀺𑀮𑀸𑀖𑀸𑀬𑀁 𑀫𑀣-𑀫𑀢𑁆𑀣 𑀯𑀺𑀮𑁄𑀴𑀦𑁂,

𑀦𑀸𑀣 𑀬𑀸𑀘𑀦𑀲𑀦𑁆𑀢𑀸𑀧 𑀇𑀲𑁆𑀲𑁂𑀭𑀸’𑀲𑀺𑀁𑀲𑀦𑁂𑀲𑀼 (𑀘;)

𑁩𑁨.

𑀧𑀼𑀣 (𑀘𑁂) 𑀧𑀼𑀣𑀼 𑀯𑀺𑀢𑁆𑀣𑀸𑀭𑁂 𑀩𑁆𑀬𑀣 𑀪𑀻𑀢𑀺𑀘𑀮𑁂𑀲𑀼 (𑀘),

𑀕𑁄𑀢𑁆𑀣𑀼 𑀯𑀁𑀲𑁂 𑀧𑀣-𑀧𑀦𑁆𑀣 𑀕𑀢𑁂 𑀦𑀦𑁆𑀤 𑀲𑀫𑀺𑀤𑁆𑀥𑀺𑀬𑀁;

𑁩𑁩.

𑀯𑀦𑁆𑀤𑀸’𑀪𑀺𑀯𑀸𑀤𑀣𑁄𑀫𑁂𑀲𑀼 𑀕𑀤 𑀩𑁆𑀬𑀢𑁆𑀢𑀯𑀘𑁂’(𑀧𑀺𑀘),

(𑀅𑀣𑁄) 𑀦𑀺𑀦𑁆𑀤 𑀕𑀭𑀳𑀸𑀬𑀁 𑀔𑀤𑀺 𑀧𑀓𑁆𑀔𑀦𑁆𑀤𑀦𑀸𑀤𑀺𑀲𑀼;

𑁩𑁪. ¶

𑀏𑀤𑀻 (𑀢𑀼) 𑀓𑀺𑀜𑁆𑀘𑀺𑀘𑀮𑁂𑀦 𑀘𑀤𑀺 𑀓𑀦𑁆𑀢𑀺𑀳𑀺𑀴𑀸𑀤𑀦𑁂,

𑀓𑀺𑀮𑀺𑀤𑀻 𑀧𑀭𑀺𑀤𑁂𑀯𑀸𑀤𑁄 𑀉𑀤𑀺𑀲𑁆𑀲𑀯𑀓𑀺𑀮𑁂𑀤𑀦𑁂;

𑁩𑁫.

𑀇𑀤𑀻 (𑀢𑀼) 𑀧𑀭𑀫𑀺𑀲𑁆𑀲𑀭𑀺𑀬𑁂 𑀅𑀤𑀺𑀅𑀦𑁆𑀤𑀼 (𑀘) 𑀩𑀦𑁆𑀥𑀦𑁂,

𑀪𑀕𑀦𑁆𑀤 𑀲𑁂𑀯𑀦𑁂 (𑀳𑁄𑀢𑀺) 𑀪𑀤𑁆𑀤 𑀓𑀮𑁆𑀬𑀸𑀡𑀓𑀫𑁆𑀫𑀦𑀺;

𑁩𑁬.

𑀲𑀺𑀤 𑀲𑀺𑀗𑁆𑀕𑀸𑀭𑀧𑀸𑀓𑁂𑀲𑀼 𑀲𑀤𑁆𑀤𑀼𑀳𑀭𑀺𑀢𑀲𑁄𑀲𑀦𑁂,

𑀫𑀤𑀺 𑀩𑀮𑁆𑀬𑁂 𑀫𑀼𑀤-𑀫𑀤𑀸 𑀲𑀦𑁆𑀢𑁄𑀲𑁂 𑀫𑀤𑁆𑀤 𑀫𑀤𑁆𑀤𑀦𑁂;

𑁩𑁭.

𑀲𑀦𑁆𑀤𑀼 𑀧𑀲𑁆𑀲𑀯𑀦𑀸𑀤𑀻𑀲𑀼 𑀓𑀦𑁆𑀤’𑀯𑁆𑀳𑀸𑀦𑁂 (𑀘’) 𑀭𑁄𑀤𑀦𑁂,

𑀯𑀺𑀤 𑀮𑀸𑀪𑁂 𑀤𑀤 𑀤𑀸𑀦𑁂 𑀭𑀼𑀤𑀺 𑀅𑀲𑁆𑀲𑀼𑀯𑀺𑀫𑁄𑀘𑀦𑁂;

𑁩𑁮.

𑀲𑀤𑁄 𑀯𑀺𑀲𑀭𑀡𑀸’𑀤𑀸𑀦𑀕𑀫𑀦𑁂 (𑀘𑀸’)𑀯𑀲𑀸𑀤𑀦𑁂,

𑀳𑀺𑀴𑀸𑀤 (𑀢𑀼) 𑀲𑀼𑀔𑁂 𑀲𑀽𑀤𑀓𑁆𑀔𑀭𑀡𑁂 𑀭𑀤 𑀯𑀺𑀮𑁂𑀔𑀡𑁂;

𑁩𑁯.

𑀲𑀸𑀤 𑀅𑀲𑁆𑀲𑀸𑀤𑀦𑀸𑀤𑀻𑀲𑀼 𑀕𑀤 𑀩𑁆𑀬𑀢𑁆𑀢𑀯𑀘𑁂’(𑀧𑀺𑀘),

𑀦𑀤 𑀅𑀩𑁆𑀬𑀢𑁆𑀢𑀲𑀤𑁆𑀤𑁂 (𑀢𑀼) 𑀭𑀤𑀸’𑀤𑀸-𑀔𑀸𑀤-𑀪𑀓𑁆𑀔𑀡𑁂;

𑁪𑁦. ¶

𑀅𑀤𑁆𑀤 𑀬𑀸𑀘𑀦𑀬𑀸𑀢𑁆𑀭𑀸𑀤𑀺𑀲𑁆𑀯 (𑀣𑁄) 𑀫𑀺𑀤 𑀲𑀺𑀦𑁂𑀳𑀦𑁂,

(𑀲𑀺𑀬𑀸) 𑀔𑀼𑀤 𑀚𑀺𑀕𑀘𑁆𑀙𑀸𑀬𑀁 𑀤𑀴𑀺𑀤𑁆𑀤 𑀤𑀼𑀕𑁆𑀕𑀘𑁆𑀘𑀁 (𑀳𑀺 𑀢𑀼;)

𑁪𑁧.

𑀤𑀸 𑀤𑀯𑁂 𑀤𑀼 𑀕𑀢𑀻𑀯𑀼𑀤𑁆𑀥𑀬𑀁 𑀤𑀸 𑀤𑀸𑀦𑁂 𑀯𑀺𑀤 𑀚𑀸𑀦𑀦𑁂,

𑀢𑀤𑀺 𑀆𑀮𑀲𑀺𑀬𑁂 𑀩𑀸𑀥 𑀩𑀸𑀥𑀸𑀬𑀁 𑀕𑀼𑀥 𑀓𑀻𑀴𑀦𑁂;

𑁪𑁨.

(𑀅𑀣𑁄) 𑀕𑀸𑀥 𑀧𑀢𑀺𑀝𑁆𑀞𑀸𑀬𑀁 𑀯𑀼𑀞𑀼-𑀏𑀥 (𑀘) 𑀯𑀼𑀤𑁆𑀥𑀺𑀬𑀁,

𑀥𑀸 (𑀳𑁄𑀢𑀺) 𑀥𑀸𑀭𑀡𑁂 (𑀘𑁂𑀯) 𑀘𑀺𑀦𑁆𑀢𑀸𑀬𑀁 𑀩𑀼𑀥 𑀩𑁄𑀥𑀦𑁂;

𑁪𑁩.

𑀲𑀺𑀥𑀼 𑀕𑀢𑀺𑀫𑁆𑀳𑀺 𑀬𑀼𑀥 𑀲𑀫𑁆𑀧𑀳𑀸𑀭𑁂 𑀯𑀺𑀥 𑀯𑁂𑀥𑀦𑁂,

𑀭𑀸𑀥 𑀳𑀺𑀁𑀲𑀸𑀬𑀲𑀁𑀭𑀸𑀥𑁂 𑀩𑀥-𑀩𑀦𑁆𑀥 (𑀘) 𑀩𑀦𑁆𑀥𑀦𑁂;

𑁪𑁪.

𑀲𑀺𑀥-𑀲𑀸𑀥 (𑀘) 𑀲𑀺𑀤𑁆𑀥𑀺𑀫𑁆𑀳𑀺 𑀥𑁂 𑀧𑀸𑀦𑁂 𑀇𑀦𑁆𑀥 𑀤𑀺𑀢𑁆𑀢𑀺𑀬𑀁,

𑀫𑀸𑀦 𑀧𑀽𑀚𑀸𑀬 𑀯𑀦-𑀲𑀦 𑀲𑀫𑁆𑀪𑀯𑁂 𑀅𑀦 𑀧𑀸𑀡𑀦𑁂;

𑀓𑀦 𑀤𑀺𑀢𑁆𑀢𑀺𑀕𑀢𑀻𑀓𑀦𑁆𑀢𑁆𑀬𑀁 𑀔𑀦-𑀔𑀦𑁆𑀯’𑀯𑀤𑀸𑀭𑀡𑁂.

𑁪𑁫. ¶

𑀕𑀼𑀧 𑀕𑁄𑀧𑀦𑀓𑁂 𑀕𑀼𑀧 𑀲𑀁𑀯𑀭𑀡𑁂 𑀢𑀧 𑀲𑀦𑁆𑀢𑀸𑀧𑁂 𑀢𑀧 𑀇𑀲𑁆𑀲𑀭𑀺𑀬𑁂,

𑀘𑀼𑀧 𑀫𑀦𑁆𑀤𑀕𑀢𑁂 𑀢𑀧𑀼𑀉𑀩𑁆𑀩𑁂𑀕𑁂 𑀭𑀧-𑀮𑀧 𑀯𑀸𑀓𑁆𑀬𑁂 𑀲𑀧 𑀅𑀓𑁆𑀓𑁄𑀲𑁂;

𑁪𑁬.

𑀚𑀧-𑀚𑀧𑁆𑀧 𑀯𑀘𑁂’𑀩𑁆𑀬𑀢𑁆𑀢𑁂 𑀢𑀧𑁆𑀧 𑀲𑀦𑁆𑀢𑀧𑁆𑀧𑀦𑁂 (𑀲𑀺𑀬𑀸),

𑀓𑀧𑀺 𑀓𑀺𑀜𑁆𑀘𑀺𑀘𑀮𑁂 𑀓𑀧𑁆𑀧 𑀲𑀸𑀫𑀢𑁆𑀣𑁂 𑀯𑁂𑀧𑀼 𑀓𑀫𑁆𑀧𑀦𑁂;

𑁪𑁭.

𑀢𑀧𑁆𑀧 𑀲𑀦𑁆𑀢𑀕𑀢𑁂𑀘𑁆𑀙𑁂𑀤𑁂 𑀢𑀓𑁆𑀓𑁂 𑀳𑀺𑀁𑀲𑀸𑀤𑀺𑀲𑀼’(𑀘𑁆𑀘𑀢𑁂),

𑀯𑀧 𑀩𑀻𑀚𑀯𑀺𑀦𑀺𑀓𑁆𑀔𑁂𑀧𑁂 𑀥𑀽𑀧 𑀲𑀦𑁆𑀢𑀧𑀦𑁂’(𑀧𑀺 𑀘);

𑁪𑁮.

𑀘𑀧 𑀲𑀸𑀦𑁆𑀢𑁆𑀯𑁂 𑀧𑀼 𑀧𑀯𑀦𑁂 𑀛𑀧 𑀤𑀸𑀳𑁂 𑀲𑀼𑀧𑁄 𑀲𑀬𑁂,

𑀧𑀼𑀧𑁆𑀨 𑀯𑀺𑀓𑀲𑀦𑁂 (𑀳𑁄𑀢𑀺) 𑀭𑀫𑁆𑀩’𑀮𑀫𑁆𑀩𑀯𑀲𑀁𑀲𑀦𑁂;

𑁪𑁯.

𑀘𑀼𑀫𑁆𑀩 𑀯𑀤𑀦𑀲𑀁𑀬𑁄𑀕𑁂 𑀓𑀫𑁆𑀩 𑀲𑀁𑀯𑀭𑀡𑁂 (𑀫𑀢𑁄),

𑀅𑀫𑁆𑀩 𑀲𑀤𑁆𑀤𑁂 (𑀘) 𑀅𑀲𑁆𑀲𑀸𑀤𑁂 𑀢𑀸𑀬𑀦𑁂 𑀲𑀩𑀺 𑀫𑀡𑁆𑀟𑀦𑁂;

𑁫𑁦. ¶

𑀕𑀩𑁆𑀩 𑀤𑀧𑁆𑀧𑁂’𑀩𑁆𑀩-𑀲𑀩𑁆𑀩𑀸’(𑀧𑀺) 𑀕𑀫𑀦𑁂 𑀧𑀼𑀩𑁆𑀩 𑀧𑀽𑀭𑀡𑁂,

𑁫𑁧.

𑀮𑀪 𑀮𑀸𑀪𑁂 𑀚𑀫𑁆𑀪 𑀕𑀢𑁆𑀢𑀯𑀺𑀦𑀸𑀫𑁂 𑀲𑀼𑀪 𑀲𑁄𑀪𑀦𑁂,

𑀪𑀻 𑀪𑀬𑁂 𑀭𑀪 𑀭𑀸𑀪𑀲𑁆𑀲𑁂 (𑀘𑀸)’𑀭𑀫𑁆𑀪𑁂 𑀔𑀼𑀪 𑀲𑀜𑁆𑀘𑀮𑁂;

𑁫𑁨.

𑀣𑀫𑁆𑀪-𑀔𑀫𑁆𑀪 𑀧𑀢𑀺𑀩𑀦𑁆𑀥𑁂 𑀕𑀩𑁆𑀪 𑀧𑀸𑀕𑀩𑁆𑀪𑀺𑀬𑁂 𑀯𑀥𑁂,

𑀲𑀼𑀫𑁆𑀪 𑀲𑀁𑀲𑀼𑀫𑁆𑀪𑀦𑁂 𑀲𑀫𑁆𑀪 𑀯𑀺𑀲𑁆𑀲𑀸𑀲𑁂 𑀬𑀪 𑀫𑁂𑀣𑀼𑀦𑁂;

𑁫𑁩.

𑀤𑀼𑀪 𑀚𑀻𑀕𑀺𑀁𑀲𑀦𑁂 𑀤𑀩𑁆𑀪 𑀕𑀦𑁆𑀣𑀦𑁂 𑀉𑀤𑁆𑀭𑀪𑀸’𑀤𑀦𑁂,

𑀓𑀫𑀽 (𑀢𑀼) 𑀧𑀤𑀯𑀺𑀓𑁆𑀔𑁂𑀧𑁂 𑀔𑀫𑀽 (𑀢𑀼) 𑀲𑀳𑀡𑁂 (𑀲𑀺𑀬𑀸;)

𑁫𑁪.

𑀪𑀫𑀼 𑀅𑀦𑀯𑀝𑁆𑀞𑀸𑀦𑁂 (𑀘) 𑀯𑀫𑀼 𑀉𑀕𑁆𑀕𑀺𑀭𑀡𑀸𑀤𑀺𑀲𑀼,

𑀓𑀺𑀮𑀫𑀼-𑀓𑁆𑀮𑀫𑀽 𑀕𑁂𑀮𑀜𑁆𑀜𑁂 𑀭𑀫𑀼 𑀓𑀻𑀴𑀸’𑀬 (𑀫𑀻𑀭𑀺𑀢𑁄;)

𑁫𑁫.

𑀤𑀫𑁄 𑀤𑀫𑁂 𑀦𑀫 𑀦𑀫𑁂 (𑀅𑀣𑁄) 𑀲𑀫 𑀧𑀭𑀺𑀲𑁆𑀲𑀫𑁂,

𑀬𑀫𑀼 𑀉𑀧𑀭𑀫𑁂 𑀦𑀸𑀲𑁂 𑀅𑀫 𑀬𑀸𑀢𑁂 𑀫𑀼 𑀩𑀦𑁆𑀥𑀦𑁂;

𑁫𑁬. ¶

𑀥𑀫𑁄 𑀧𑀼𑀫𑁄 (𑀘) 𑀥𑀫𑀦𑁂 𑀢𑀫 𑀲𑀗𑁆𑀓𑀸𑀯𑀺𑀪𑀽𑀲𑀦𑁂,

𑀥𑀼𑀫-𑀣𑀻𑀫 (𑀘) 𑀲𑀗𑁆𑀖𑀸𑀢𑁂 𑀢𑀫 𑀲𑀸𑀦𑁆𑀢𑁆𑀯’𑀯𑀲𑀸𑀤𑀺𑀬𑁂;

𑁫𑁭.

𑀅𑀬𑁄 𑀯𑀬𑁄 𑀧𑀬-𑀫𑀬𑁄 𑀦𑀬𑁄 𑀭𑀬𑀕𑀢𑀺𑀫𑁆𑀳𑀺 (𑀘)

𑀤𑀬 𑀤𑀸𑀦𑀕𑀢𑀻𑀭𑀓𑁆𑀔𑀸 𑀳𑀺𑀁𑀲𑀸𑀤𑀺𑀲𑀼 𑀬𑀼 𑀫𑀺𑀲𑁆𑀲𑀦𑁂;

𑀘𑀸𑀬 𑀲𑀫𑁆𑀧𑀽𑀚𑀦𑁂 𑀢𑀸𑀬 𑀲𑀦𑁆𑀢𑀸𑀦𑁂 𑀧𑀸𑀬 𑀯𑀼𑀤𑁆𑀥𑀺𑀬𑀁,

(𑀅𑀣𑁄) 𑀉𑀲𑀽𑀬 𑀤𑁄𑀲𑀸’𑀯𑀺𑀓𑀭𑀡𑁂 𑀲𑀸𑀬 𑀲𑀸𑀬𑀦𑁂;

𑁫𑁮.

𑀢𑀭 𑀢𑀭𑀡𑀲𑁆𑀫𑀺𑀁 𑀣𑀭 𑀲𑀦𑁆𑀣𑀭𑀡𑁂 𑀪𑀭 𑀪𑀭𑀡𑀲𑁆𑀫𑀺𑀁 𑀨𑀭 𑀲𑀫𑁆𑀨𑀭𑀡𑁂,

𑀲𑀭 𑀕𑀢𑀺 𑀘𑀺𑀦𑁆𑀢𑀸 𑀳𑀺𑀁𑀲𑀸 𑀲𑀤𑁆𑀤𑁂 𑀨𑀼𑀭 𑀘𑀮𑀦𑀸𑀤𑁄 𑀳𑀭 𑀳𑀭𑀡𑀲𑁆𑀫𑀺𑀁;

𑁫𑁯.

𑀭𑀺 𑀲𑀦𑁆𑀢𑀢𑀺𑀲𑁆𑀫𑀺𑀁 𑀭𑀺 𑀕𑀢𑁂 𑀭𑀼 𑀲𑀤𑁆𑀤𑁂 𑀔𑀼𑀭𑀘𑁆𑀙𑀺𑀤𑀲𑁆𑀫𑀺𑀁 𑀥𑀭 𑀥𑀸𑀭𑀡𑀫𑁆𑀳𑀺,

𑀚𑀭 𑀚𑀻𑀭𑀡𑀢𑁆𑀣𑁂 𑀫𑀭𑀧𑀸𑀡𑀘𑀸𑀕𑁂 𑀔𑀭 𑀲𑁂𑀓𑀦𑀸𑀲𑁂 𑀖𑀭 𑀲𑁂𑀯𑀦𑀫𑁆𑀳𑀺;

𑁬𑁦. ¶

𑀕𑀭𑁄 𑀦𑀺𑀕𑀭𑁂𑀡 𑀲𑁂𑀓𑁂 𑀤𑀭 𑀟𑀸𑀳𑁂 𑀯𑀺𑀤𑀸𑀭𑀡𑁂,

𑀘𑀭 𑀕𑀢𑀺𑀪𑀓𑁆𑀔𑀡𑁂𑀲𑀼 𑀯𑀭 𑀲𑀁𑀯𑀭𑀡𑀸𑀤𑀺𑀲𑀼;

𑁬𑁧.

𑀘𑀭𑀘𑁆𑀙𑁂𑀤𑁂 𑀅𑀭𑀦𑀸𑀲𑁂 𑀕𑀢𑁂 (𑀘) 𑀧𑀽𑀭 𑀧𑀽𑀭𑀡𑁂,

𑀓𑀼𑀭 𑀓𑁆𑀓𑁄𑀲𑁂 𑀦𑀭 𑀦𑀬𑁂 𑀚𑀸𑀕𑀭 𑀲𑀼𑀧𑀺𑀦𑀓𑁆𑀔𑀬𑁂;

𑁬𑁨.

𑀧𑀻𑀮𑀼-𑀧𑀮𑀽-𑀲𑀮-𑀳𑀼𑀮𑀸 𑀕𑀢𑁆𑀬’𑀢𑁆𑀣𑀸 𑀘𑀮 𑀓𑀫𑁆𑀧𑀦𑁂,

𑀔𑀮 𑀲𑀜𑁆𑀘𑀮𑀦𑁂 𑀨𑀼𑀮𑁆𑀮 𑀯𑀺𑀓𑀸𑀲𑁂 𑀚𑀮 𑀤𑀺𑀢𑁆𑀢𑀺𑀬𑀁;

𑁬𑁩.

𑀨𑀮 𑀦𑀺𑀧𑁆𑀨𑀢𑁆𑀢𑀺𑀬𑀁 (𑀳𑁄𑀢𑀺) 𑀤𑀮 𑀤𑀺𑀢𑁆𑀢𑀺𑀯𑀺𑀤𑀸𑀭𑀡𑁂,

𑀤𑀮 𑀤𑀼𑀕𑁆𑀕𑀢𑀺𑀬𑀁 𑀦𑀻𑀮 𑀯𑀡𑁆𑀡𑁂 𑀫𑀻𑀮 𑀦𑀺𑀫𑀻𑀮𑀦𑁂;

𑁬𑁪.

𑀲𑀺𑀮 𑀲𑀫𑀸𑀥𑀺𑀫𑁆𑀳𑀺 𑀓𑀻𑀮 𑀩𑀦𑁆𑀥𑁂 𑀕𑀮-𑀕𑀺𑀮𑀸’𑀤𑀦𑁂,

𑀓𑀽𑀮 𑀆𑀯𑀭𑀡𑁂 𑀲𑀽𑀮 𑀭𑀼𑀚𑀸𑀬𑀁 𑀩𑀮𑀧𑀸𑀡𑀦𑁂;

𑁬𑁫.

𑀢𑀮-𑀫𑀽𑀮 𑀧𑀢𑀺𑀝𑁆𑀞𑀸𑀬𑀁 𑀯𑀮-𑀯𑀮𑁆𑀮 𑀦𑀺𑀯𑀸𑀭𑀡𑁂,

𑀧𑀮𑁆𑀮 𑀦𑀺𑀦𑁆𑀦𑁂 (𑀘) 𑀕𑀫𑀦𑁂 𑀫𑀮-𑀫𑀮𑁆𑀮’𑀯𑀥𑀸𑀭𑀡𑁂;

𑁬𑁬. ¶

(𑀯𑀢𑁆𑀢𑀢𑁂) 𑀔𑀺𑀮 𑀓𑀸𑀞𑀺𑀦𑁆𑀦𑁂 𑀓𑀮𑀺𑀮𑁂 𑀅𑀮-𑀓𑀮 𑀤𑁆𑀯𑀬𑀁,

𑀯𑁂𑀮𑁆𑀮 𑀲𑀜𑁆𑀘𑀮𑀦𑁂 𑀓𑀮𑁆𑀮 𑀲𑀚𑁆𑀚𑀦𑁂 𑀅𑀮𑀺𑀩𑀦𑁆𑀥𑀦𑁂;

𑁬𑁭.

𑀘𑀼𑀮𑁆𑀮 𑀳𑀸𑀯𑀓𑀺𑀭𑀬𑁂 𑀣𑀽𑀮𑀸’𑀓𑀲𑁆𑀲𑀦𑁂 𑀘𑀽𑀮 𑀫𑀤𑁆𑀤𑀦𑁂,

(𑀯𑀢𑁆𑀢𑀢𑁂) 𑀔𑀮 𑀲𑁄𑀘𑁂𑀬𑁆𑀬𑁄 𑀧𑀮 𑀭𑀓𑁆𑀔𑀕𑀢𑁂𑀲𑀼(𑀧𑀺;)

𑁬𑁮.

𑀓𑁂𑀮-𑀔𑁂𑀮-𑀘𑁂𑀮-𑀧𑁂𑀮-𑀯𑁂𑀮-𑀲𑀜𑁆𑀘𑀮𑀦𑀸𑀤𑀺𑀲𑀼,

𑀅𑀯 𑀭𑀓𑁆𑀔𑀡𑁂 𑀚𑀻𑀯 𑀧𑀸𑀡𑀥𑀸𑀭𑀡𑁂 (𑀢𑀼) 𑀧𑁆𑀮𑀯𑁄 𑀕𑀢𑁂;

𑁬𑁯.

𑀓𑀡𑁆𑀟𑀼𑀯𑀦𑀫𑁆𑀳𑀺 𑀓𑀡𑁆𑀟𑀼𑀯𑁄 𑀲𑀭𑀡𑁂 𑀙𑁂𑀤𑀦𑁂 𑀤𑀯𑁂,

𑀤𑀯𑁄 (𑀢𑀼) 𑀤𑀯𑀦𑁂 𑀤𑁂𑀯𑀼 𑀤𑁂𑀯𑀦𑁂 𑀲𑁂𑀯𑀼 𑀲𑁂𑀯𑀦𑁂;

𑁭𑁦.

𑀥𑀸𑀯 𑀕𑀫𑀦𑀯𑀼𑀤𑁆𑀥𑀺𑀫𑁆𑀳𑀺 (𑀧𑀞𑀺𑀢𑁄) 𑀥𑁄𑀯𑀼 𑀥𑁄𑀯𑀦𑁂;

𑀯𑁂-𑀯𑀻 𑀤𑁆𑀯𑁂 𑀢𑀦𑁆𑀢𑀼𑀲𑀦𑁆𑀢𑀸𑀦𑁂 𑀯𑁂-𑀯𑀼 𑀲𑀁𑀯𑀭𑀡𑁂 (𑀲𑀺𑀬𑀸)

𑀳𑁆𑀯𑁂 𑀅𑀯𑁆𑀳𑀸𑀦𑁂 𑀓𑁂𑀯 𑀲𑁂𑀓𑁂 𑀥𑀼𑀯 𑀬𑀸𑀢𑁆𑀭𑀸 𑀣𑀺𑀭𑁂𑀲𑀼 (𑀘;);

𑁭𑁧. ¶

𑀅𑀲 𑀕𑀲 𑀅𑀤𑀦𑁂 𑀖𑀲 𑀅𑀤𑀦𑀲𑁆𑀫𑀺𑀁-𑀇𑀲 𑀧𑀭𑀺𑀬𑁂𑀲𑁂 𑀇𑀲𑀼𑀇𑀘𑁆𑀙𑀸𑀬𑀁,

𑀲𑀲𑀼 𑀧𑀸𑀡𑀦𑀕𑀢𑀺𑀳𑀺𑀁𑀲𑀸’𑀤𑁆𑀬’𑀢𑁆𑀣𑁂-𑀫𑀲 𑀆𑀫𑀲𑀦𑁂 𑀫𑀼𑀲 𑀲𑀫𑁆𑀫𑁄𑀲𑁂;

𑁭𑁨.

𑀓𑀼𑀲 𑀅𑀓𑁆𑀓𑁄𑀲𑁂 𑀤𑀼𑀲 𑀅𑀧𑁆𑀧𑀻𑀢𑁂-𑀢𑀼𑀲 𑀲𑀦𑁆𑀢𑁄𑀲𑁂 𑀧𑀼𑀲 𑀧𑁄𑀲𑀫𑁆𑀳𑀺,

𑀭𑀼𑀲 𑀆𑀮𑁂𑀧𑁂 𑀭𑀼𑀲 𑀳𑀺𑀁𑀲𑀸𑀬𑀁-𑀫𑀲𑀼 𑀫𑀘𑁆𑀙𑁂𑀭𑁂 𑀉𑀲𑀼 𑀤𑀸𑀳𑁂 (’𑀧𑀺;)

𑁭𑁩.

𑀳𑀲 𑀳𑀲𑀦𑀲𑁆𑀫𑀺𑀁 𑀖𑀼𑀲 𑀲𑀤𑁆𑀤𑀲𑁆𑀫𑀺𑀁-𑀢𑀲 𑀉𑀩𑁆𑀩𑁂𑀕𑁂 𑀢𑁆𑀭𑀲 𑀉𑀩𑁆𑀩𑁂𑀕𑁂,

𑀮𑀲 𑀓𑀦𑁆𑀢𑁆𑀬’𑀢𑁆𑀣𑁂 𑀭𑀲 𑀅𑀲𑁆𑀲𑀸𑀤𑁂-(𑀧𑀼𑀦)𑀪𑀲 𑀪𑀲𑁆𑀫𑀺𑀓𑀭𑀡𑁂(𑀘𑀸’𑀧𑀺;)

𑁭𑁪.

𑀕𑀯𑁂𑀲 𑀫𑀕𑁆𑀕𑀡𑁂 𑀧𑀁𑀲 𑀦𑀸𑀲𑀦𑁂 𑀤𑀺𑀲 𑀧𑁂𑀓𑁆𑀔𑀡𑁂,

𑀲𑀸𑀲𑀸’𑀦𑀼𑀲𑀺𑀝𑁆𑀞𑀺𑀬𑀁 𑀳𑀁𑀲 𑀧𑀺𑀢𑀺𑀬𑀁 𑀧𑀸𑀲 𑀩𑀦𑁆𑀥𑀦𑁂;

𑁭𑁫.

𑀲𑀁𑀲 𑀧𑀲𑀁𑀲𑀦𑁂 𑀇𑀲𑁆𑀲 𑀇𑀲𑁆𑀲𑀸𑀬𑀁 𑀓𑀲𑁆𑀲 𑀓𑀲𑁆𑀲𑀦𑁂,

𑀥𑀁𑀲 𑀧𑀥𑀁𑀲𑀦𑁂 𑀲𑀺𑀁𑀲 𑀇𑀘𑁆𑀙𑀸𑀬𑀁 𑀖𑀁𑀲 𑀖𑀁𑀲𑀦𑁂;

𑁭𑁬. ¶

𑀲𑀁𑀲-𑀤𑀁𑀲𑀸 (𑀢𑀼) 𑀟𑀲𑀦𑁂 𑀪𑀸𑀲 𑀯𑀸𑀘𑀸𑀬 𑀤𑀺𑀢𑁆𑀢𑀺𑀬𑀁,

𑀲𑀺𑀯𑀼 𑀢𑀦𑁆𑀢𑀽𑀦𑀲𑀦𑁆𑀢𑀸𑀦𑁂 𑀔𑀻 𑀔𑀬𑁂 𑀔𑀸 𑀧𑀓𑀸𑀲𑀦𑁂;

𑁧𑁦𑁪.

𑀓𑀸-𑀕𑀸 𑀲𑀤𑁆𑀤𑁂 (𑀧𑀺) 𑀖𑀸 𑀕𑀦𑁆𑀥𑁄’𑀧𑀸𑀤𑀸𑀦𑁂 𑀭𑀼𑀘 𑀭𑁄𑀘𑀦𑁂,

𑀓𑀘 𑀤𑀺𑀢𑁆𑀬𑀁 𑀫𑀼𑀘 𑀫𑁄𑀘𑁂 (𑀅𑀣𑁄) 𑀯𑀺𑀘 𑀯𑀺𑀯𑁂𑀘𑀦𑁂;

𑁧𑁦𑁫.

𑀭𑀜𑁆𑀚 𑀭𑀸𑀕𑁂 𑀲𑀜𑁆𑀚 𑀲𑀗𑁆𑀕𑁂 𑀔𑀮𑀦𑁂 𑀫𑀚𑁆𑀚 𑀲𑀼𑀤𑁆𑀥𑀺𑀬𑀁,

𑀬𑀼𑀚𑁄 𑀲𑀫𑀸𑀥𑀺𑀫𑁆𑀳𑀺 𑀮𑀼𑀚𑁄 𑀯𑀺𑀦𑀸𑀲𑁂 𑀛𑀸 𑀯𑀺𑀘𑀺𑀦𑁆𑀢𑀦𑁂;

𑁧𑁦𑁬.

𑀢𑀸 𑀧𑀸𑀮𑀦𑁂 𑀙𑀺𑀤𑀺 𑀤𑁆𑀯𑁂𑀥𑀸𑀓𑀸𑀭𑁂 𑀫𑀺𑀤 𑀲𑀺𑀦𑁂𑀳𑀦𑁂,

𑀫𑀤𑀼’𑀫𑁆𑀫𑀸𑀤𑁂 𑀔𑀺𑀤 𑀤𑀻𑀦𑀪𑀸𑀯𑁂 𑀪𑀺𑀤 𑀯𑀺𑀤𑀸𑀭𑀡𑁂;

𑁧𑁦𑁭.

𑀲𑀺𑀤 𑀧𑀸𑀓𑁂 𑀧𑀤𑀕𑀢𑁂 𑀯𑀺𑀤 𑀲𑀢𑁆𑀢𑀸 𑀯𑀺𑀘𑀺𑀦𑁆𑀢𑀦𑁂,

𑀤𑀻 𑀔𑀬𑁂 𑀲𑀼𑀧𑀦𑁂 𑀤𑀸 (𑀘) 𑀤𑀸𑀦𑁂 𑀤𑀸𑀢𑁆𑀯’𑀯𑀔𑀡𑁆𑀟𑀦𑁂;

𑁧𑁦𑁮. ¶

𑀩𑀼𑀥𑀸’𑀯𑀕𑀫𑀦𑀸’𑀤𑀻𑀲𑀼 𑀅𑀢𑁆𑀣𑁂𑀲𑀼 𑀬𑀼𑀥 𑀬𑀼𑀚𑁆𑀛𑀦𑁂,

𑀓𑀼𑀥 𑀓𑁄𑀧𑁂 𑀲𑀼𑀥 𑀲𑁄𑀘𑁂 𑀭𑀸𑀥 𑀳𑀺𑀁𑀲𑀸𑀬 𑀲𑀺𑀤𑁆𑀥𑀺𑀬𑀁;

𑁧𑁦𑁯.

𑀇𑀥 𑀲𑀁𑀲𑀺𑀤𑁆𑀥𑀺𑀯𑀼𑀤𑁆𑀥𑀻𑀲𑀼 𑀲𑀺𑀥-𑀲𑀸𑀥 (𑀘) 𑀲𑀺𑀤𑁆𑀥𑀺𑀬𑀁,

𑀯𑀺𑀥 𑀯𑁂𑀥𑁂 𑀕𑀺𑀥 𑀕𑁂𑀥𑁂 𑀭𑀼𑀥𑀺 𑀆𑀯𑀭𑀡𑀸’𑀤𑀺𑀲𑀼;

𑁧𑁧𑁦.

𑀫𑀦 𑀜𑀸𑀡𑁂 𑀚𑀦𑀼’𑀧𑁆𑀧𑀸𑀤𑁂 𑀳𑀦 𑀳𑀺𑀁𑀲𑀸𑀕𑀢𑀻𑀲𑀼 (𑀧𑀺,)

𑀲𑀺𑀦𑀸 𑀲𑁄𑀘𑁂 𑀓𑀼𑀧 𑀓𑁄𑀧𑁂 𑀢𑀧 𑀲𑀦𑁆𑀢𑀸𑀧 𑀧𑀻𑀡𑀦𑁂;

𑁧𑁧𑁧.

𑀮𑀼𑀧𑀘𑁆𑀙𑁂𑀤𑁂 𑀭𑀼𑀧 𑀦𑀸𑀲𑁂 𑀧𑀓𑀸𑀲𑁂 𑀤𑀺𑀧 𑀤𑀺𑀢𑁆𑀢𑀺𑀬𑀁,

𑀤𑀧 𑀳𑀸𑀲𑁂 𑀮𑀪 𑀮𑀸𑀪𑁂 𑀮𑀼𑀪 𑀕𑁂𑀥𑁂 𑀔𑀼𑀪𑁄 𑀘𑀮𑁂;

𑁧𑁧𑁨.

𑀲𑀫𑀽’𑀧𑀲𑀫 𑀔𑁂𑀤𑁂𑀲𑀼 𑀳𑀭-𑀳𑀺𑀭𑀻 (𑀘) 𑀮𑀚𑁆𑀚𑀦𑁂,

𑀫𑀺𑀮𑀸 𑀕𑀢𑁆𑀢𑀯𑀻𑀦𑀸𑀫𑁂 (𑀘) 𑀕𑀺𑀮𑀸 𑀳𑀸𑀲𑀓𑁆𑀔𑀬𑁂 (𑀧𑀺 𑀘;)

𑁧𑁧𑁩. ¶

𑀮𑀻 𑀲𑀺𑀮𑁂𑀲𑁂 𑀤𑁆𑀭𑀯𑀻𑀓𑀸𑀭𑁂 𑀯𑀸 𑀕𑀢𑀻 𑀩𑀦𑁆𑀥𑀦𑁂𑀲𑀼 (𑀘,)

𑀮𑀺𑀲𑀺 𑀮𑁂𑀲𑁂 𑀢𑀼𑀲 𑀢𑁄𑀲𑁂 𑀲𑀺𑀮𑀺𑀲𑀸’𑀮𑀺𑀗𑁆𑀕𑀦𑀸𑀤𑀺𑀲𑀼;

𑁧𑁧𑁪.

𑀓𑀺𑀮𑀺𑀲 𑀓𑀮𑀺𑀲𑁄’𑀧𑀢𑀸𑀧𑁂 (𑀅𑀣𑁄) 𑀢𑀲 𑀧𑀺𑀧𑀸𑀲𑀦𑁂,

𑀭𑀼𑀲 𑀭𑁄𑀲𑁂 𑀤𑀺𑀲-𑀤𑀼𑀲 𑀅𑀧𑁆𑀧𑀻𑀢𑀺𑀫𑁆𑀳𑀺 (𑀤𑀼𑀯𑁂 𑀲𑀺𑀬𑀼𑀁;)

𑁧𑁧𑁫.

𑀬𑀲𑀼𑀧𑁆𑀧𑀬𑀢𑀦𑁂 𑀅𑀲𑀼 𑀔𑁂𑀧𑀦𑁂 (𑀧𑀺 𑀘 𑀯𑀢𑁆𑀢𑀢𑁂,)

𑀲𑀼𑀲 𑀲𑁄𑀲𑁂 𑀪𑀲 𑀅𑀥𑁄𑀧𑀸𑀢𑁂 𑀦𑀲 𑀅𑀤𑀲𑁆𑀲𑀦𑁂;

𑁧𑁧𑁬. 𑀲𑀸’𑀲𑁆𑀲𑀸𑀤𑁂 𑀲𑀸’𑀯𑀲𑀸𑀦𑁂 (𑀘) 𑀲𑀸 𑀢𑀦𑀽𑀓𑀭𑀡𑁂 (𑀧𑀺 𑀘) 𑀳𑀸 𑀘𑀸𑀕𑁂 𑀫𑀼𑀳 𑀯𑁂𑀘𑀺𑀢𑁆𑀢𑁂 𑀦𑀳 𑀲𑀚𑁆𑀚𑀦𑀩𑀦𑁆𑀥𑀦𑁂 𑀦𑀳 𑀲𑁄𑀘𑁂 𑀧𑀺𑀳𑀺𑀘𑁆𑀙𑀸𑀬𑀁 𑀲𑀺𑀦𑀺𑀳-𑀲𑀦𑀺𑀳 𑀧𑀻𑀢𑀺𑀬𑀁.

𑀲𑁆𑀯𑀸𑀤𑀬𑁄 ¶

𑁧𑁧𑁭. 𑀲𑀼 𑀲𑀯𑀡𑁂 𑀲𑀓 𑀲𑀢𑁆𑀢𑀺𑀫𑁆𑀳𑀺 𑀔𑀻 𑀔𑀬𑀫𑁆𑀳𑀺 𑀕𑀺 𑀲𑀤𑁆𑀤𑀦𑁂,

𑀅𑀧-𑀲𑀫𑁆𑀪𑀽 (𑀘) 𑀧𑀸𑀧𑀼𑀡𑀦𑁂 𑀳𑀺 𑀕𑀢𑀺𑀫𑁆𑀳𑀺 𑀯𑀽 𑀲𑀁𑀯𑀭𑁂;

𑀓𑀺𑀬𑀸𑀤𑀬𑁄

𑁧𑁧𑁮.

𑀓𑀻 𑀯𑀺𑀦𑀺𑀫𑀬𑁂 𑀘𑀺 𑀘𑀬𑁂 𑀚𑀺 𑀚𑀬𑁂 𑀜𑀸’𑀯𑀩𑁄𑀥𑀦𑁂,

𑀣𑀯’𑀪𑀺𑀢𑁆𑀣𑀯𑁂 𑀓𑀫𑁆𑀧𑀦𑁂 𑀥𑀼 (𑀅𑀣𑁄) 𑀧𑀼 𑀧𑀯𑀦𑁂 (𑀲𑀺𑀬𑀸;)

𑁧𑁧𑁯.

𑀧𑀻 𑀢𑀧𑁆𑀧𑀡𑁂 𑀫𑀸 𑀧𑀫𑀸𑀡𑁂 𑀔𑀺𑀧𑀓𑁆𑀔𑁂𑀧𑁂 𑀫𑀺 𑀳𑀺𑀁𑀲𑀦𑁂,

𑀫𑀺 𑀧𑀫𑀸𑀡𑁂 𑀫𑀼 𑀩𑀦𑁆𑀥𑁂 (𑀘) 𑀮𑀼 𑀧𑀘𑁆𑀙𑁂𑀤𑁂 𑀲𑀺 𑀩𑀦𑁆𑀥𑀦𑁂

𑀅𑀲 𑀪𑀓𑁆𑀔𑀡𑁂 (𑀅𑀣𑁄) 𑀕𑀳 𑀉𑀧𑀸𑀤𑀸𑀦𑁂 (𑀓𑀺𑀬𑀸𑀤𑀬𑁄;)

𑀢𑀦𑀸𑀤𑀬𑁄 ¶

𑁧𑁨𑁦.

𑀢𑀦𑀼 𑀯𑀺𑀢𑁆𑀣𑀸𑀭𑁂 𑀲𑀓 𑀲𑀢𑁆𑀢𑀺𑀲𑁆𑀫𑀺𑀁-𑀤𑀼 𑀧𑀭𑀺𑀢𑀸𑀧𑁂 𑀲𑀦𑀼 𑀤𑀸𑀦𑀲𑁆𑀫𑀺𑀁,

𑀯𑀦 𑀬𑀸𑀘𑀸𑀬𑀁 𑀫𑀦𑀼 𑀩𑁄𑀥𑀲𑁆𑀫𑀺𑀁-𑀳𑀺 𑀕𑀢𑁂 𑀅𑀧 𑀧𑀸𑀧𑀼𑀡𑀦𑀲𑁆𑀫𑀺𑀁 (𑀳𑀺,)

𑀓𑀭 𑀓𑀭𑀡𑀲𑁆𑀫𑀺𑀁(𑀪𑀯𑀢𑀺)𑀲𑀺 𑀩𑀦𑁆𑀥𑁂-𑀲𑀼 𑀅𑀪𑀺𑀲𑁆𑀲𑀯𑀦𑁂(𑀢𑀦𑀼 𑀆𑀤𑀻𑀦𑀺;)

𑀦𑀺𑀘𑁆𑀘𑀁 𑀡𑁂𑀡𑀬𑀦𑁆𑀢𑀸 𑀘𑀼𑀭𑀸𑀤𑀬𑁄.

𑁧𑁨𑁧.

𑀘𑀼𑀭 𑀣𑁂𑀬𑁆𑀬𑁂 𑀮𑁄𑀓 (𑀥𑀸𑀢𑀼) 𑀤𑀲𑁆𑀲𑀦𑁂 𑀅𑀓𑀺 𑀮𑀓𑁆𑀔𑀡𑁂,

𑀲𑀺𑀬𑀸 𑀣𑀓 𑀧𑀢𑀺𑀖𑀸𑀢𑁂 (𑀧𑀼𑀦) 𑀢𑀓𑁆𑀓 𑀯𑀺𑀢𑀓𑁆𑀓𑀡𑁂;

𑁧𑁨𑁨.

𑀮𑀓𑁆𑀔 𑀤𑀲𑁆𑀲𑀦𑀅𑀗𑁆𑀓𑁂𑀲𑀼 (𑀯𑀢𑁆𑀢𑀢𑁂) 𑀫𑀓𑁆𑀔 𑀫𑀓𑁆𑀔𑀡𑁂,

𑀪𑀓𑁆𑀔𑀸’𑀤𑀦𑁂 𑀫𑁄𑀓𑁆𑀔 𑀫𑁄𑀘𑁂 𑀲𑀼𑀔-𑀤𑀼𑀓𑁆𑀔 (𑀘) 𑀢𑀓𑀺𑀭𑀬𑁂;

𑁧𑁨𑁩. ¶

𑀮𑀺𑀗𑁆𑀕 𑀘𑀺𑀢𑁆𑀢𑀓𑀺𑀭𑀬𑀸’𑀤𑀻𑀲𑀼 𑀫𑀕-𑀫𑀕𑁆𑀕 𑀕𑀯𑁂𑀲𑀦𑁂,

(𑀧𑀼𑀦𑀸’𑀧𑀺) 𑀧𑀘 𑀯𑀺𑀢𑁆𑀣𑀸𑀭𑁂 𑀓𑁆𑀮𑁂𑀲𑁂 𑀯𑀜𑁆𑀘 𑀧𑀮𑀫𑁆𑀪𑀦𑁂.

𑁧𑁨𑁪.

𑀯𑀘𑁆𑀘 𑀅𑀚𑁆𑀛𑀸𑀬𑀦𑁂 𑀅𑀘𑁆𑀘 𑀧𑀽𑀚𑀸𑀬𑀁 𑀯𑀘 𑀪𑀸𑀲𑀦𑁂,

𑀭𑀘 𑀧𑀢𑀺𑀬𑀢𑀦𑁂 𑀲𑀼𑀘 𑀧𑁂𑀲𑀼𑀜𑁆𑀜𑁂 𑀭𑀼𑀘 𑀭𑁄𑀘𑀦𑁂;

𑁧𑁨𑁫.

𑀫𑀼𑀘𑀧𑁆𑀧𑀫𑁄𑀘𑀦𑁂 𑀮𑁄𑀘 𑀤𑀲𑁆𑀲𑀦𑁂 𑀓𑀘 𑀤𑀺𑀢𑁆𑀢𑀺𑀬𑀁,

𑀲𑀚𑁆𑀚𑀸’𑀚𑁆𑀚 𑀅𑀚𑁆𑀚𑀦𑁂 𑀢𑀚𑁆𑀚 𑀢𑀚𑁆𑀚𑀦𑁂 𑀯𑀚𑁆𑀚 𑀯𑀚𑁆𑀚𑀦𑁂;

𑁧𑁨𑁬.

𑀬𑀼𑀚 𑀲𑀁𑀬𑀫𑀦𑁂 𑀧𑀽𑀚 𑀧𑀽𑀚𑀸𑀬𑀁 𑀢𑀺𑀚 𑀢𑁂𑀚𑀦𑁂,

𑀧𑀚 𑀫𑀕𑁆𑀕 𑀲𑀁𑀯𑀭𑀡𑁂 𑀕𑀢𑁂 𑀪𑀚 𑀯𑀺𑀪𑀸𑀚𑀦𑁂;

𑁧𑁨𑁭.

(𑀅𑀣𑁄) 𑀪𑀸𑀚 𑀧𑀼𑀣𑀓𑁆𑀓𑀸𑀭𑁂 𑀲𑀪𑀸𑀚 𑀧𑀻𑀢𑀺𑀤𑀲𑁆𑀲𑀦𑁂,

(𑀅𑀣𑁄 𑀢𑀼) 𑀖𑀝 𑀲𑀗𑁆𑀖𑀸𑀢𑁂 𑀖𑀝𑁆𑀝 𑀲𑀜𑁆𑀘𑀮𑀦𑀸’𑀤𑀺𑀲𑀼;

𑁧𑁨𑁮. ¶

𑀓𑀼𑀝-𑀓𑁄𑀝𑁆𑀝𑀘𑁆𑀙𑁂𑀤𑀦𑁂 (𑀤𑁆𑀯𑁂) 𑀓𑀼𑀝 𑀆𑀓𑁄𑀝𑀦𑀸’𑀤𑀺𑀲𑀼,

𑀦𑀝 𑀦𑀘𑁆𑀘𑁂 𑀘𑀝-𑀧𑀼𑀝 𑀪𑁂𑀤𑁂 𑀯𑀡𑁆𑀝 𑀯𑀺𑀪𑀸𑀚𑀦𑁂;

𑁧𑁨𑁯.

𑀢𑀼𑀯𑀝𑁆𑀝 𑀏𑀓𑀲𑀬𑀦𑁂 𑀖𑀝𑁄 𑀯𑀺𑀲𑀭𑀡𑁂 (𑀲𑀺𑀬𑀸),

𑀕𑀼𑀡𑁆𑀞 𑀑𑀕𑀼𑀡𑁆𑀞𑀦𑁂 𑀳𑁂𑀞 𑀩𑀸𑀥𑀸𑀬𑀁 𑀯𑁂𑀞 𑀯𑁂𑀞𑀦𑁂

𑀕𑀼𑀟𑀺 𑀯𑁂𑀞𑁂 𑀓𑀟𑀺-𑀔𑀟𑀺 𑀪𑁂𑀤𑀦𑁂 𑀫𑀟𑀺 𑀪𑀽𑀲𑀦𑁂;

𑁧𑁩𑁦.

𑁧𑁩𑁨.

𑀘𑀺𑀢 𑀲𑀁𑀘𑁂𑀢𑀦𑀸’𑀤𑀺𑀲𑀼 𑀓𑀺𑀢𑁆𑀢 𑀲𑀁𑀲𑀤𑁆𑀤𑀦𑁂 (𑀪𑀯𑁂;)

𑁧𑁩𑁩. ¶

𑁧𑁩𑁪.

𑁧𑁩𑁫.

𑀭𑀦𑁆𑀥 𑀧𑀸𑀓𑁂 (𑀅𑀣𑁄) 𑀫𑀸𑀦 𑀧𑀽𑀚𑀸𑀬𑀁 𑀦𑀼 𑀢𑁆𑀣𑀼𑀢𑀺𑀫𑁆𑀳𑀺 (𑀢𑀼;)

𑁧𑁩𑁬.

𑀣𑀦 𑀤𑁂𑀯𑀲𑀤𑁆𑀤𑁂 𑀊𑀦 𑀧𑀭𑀺𑀳𑀸𑀦𑁂 𑀣𑁂𑀦 𑀘𑁄𑀭𑀺𑀬𑁂,

𑀥𑀦 𑀲𑀤𑁆𑀤𑁂 𑀜𑀧 𑀢𑁄𑀲 𑀦𑀺𑀲𑀸𑀦 𑀫𑀸𑀭𑀡𑀸’𑀤𑀺𑀲𑀼;

𑁧𑁩𑁭.

𑀮𑀧 𑀯𑀸𑀓𑁆𑀬𑁂 𑀛𑀧 𑀤𑀸𑀳𑁂 𑀭𑀼𑀧 𑀭𑁄𑀧𑀡𑀆𑀤𑀺𑀲𑀼,

𑀧𑀻 𑀢𑀧𑁆𑀧𑀦𑁂 (𑀲𑀺𑀬𑀸) 𑀓𑀧𑁆𑀧 𑀯𑀺𑀢𑀓𑁆𑀓𑁂 𑀮𑀪𑀺 𑀯𑀜𑁆𑀘𑀦𑁂;

𑁧𑁩𑁮. ¶

(𑀅𑀣𑁄) 𑀯𑀳𑀺 𑀕𑀭𑀳𑀸𑀬𑀁 𑀲𑀫𑀼 𑀲𑀸𑀦𑁆𑀢𑁆𑀯𑀦 𑀤𑀲𑁆𑀲𑀦𑁂,

𑀓𑀫𑀼 𑀇𑀘𑁆𑀙𑀸𑀬 𑀓𑀦𑁆𑀢𑀺𑀫𑁆𑀳𑀺 (𑀲𑀺𑀬𑀸) 𑀣𑁄𑀫 𑀲𑀺𑀮𑀸𑀖𑀦𑁂;

𑁧𑁩𑁯.

𑀢𑀺𑀫𑀼 𑀢𑁂𑀫𑀦 𑀲𑀗𑁆𑀓𑀸𑀲𑀼 𑀅𑀫 𑀭𑁄𑀕𑀕𑀢𑀸’𑀤𑀺𑀲𑀼,

𑀲𑀁𑀕𑀸𑀫 𑀬𑀼𑀤𑁆𑀥𑁂 (𑀯𑀢𑁆𑀢𑁂𑀬𑁆𑀬) 𑀈𑀭 𑀯𑀸𑀘𑀸 𑀧𑀓𑀫𑁆𑀧𑀦𑁂;

𑁧𑁪𑁦.

𑀯𑀭 𑀆𑀯𑀭𑀡𑀺’𑀘𑁆𑀙𑀸𑀲𑀼 𑀬𑀸𑀘𑀸𑀬𑀁 𑀥𑀭 𑀥𑀸𑀭𑀡𑁂,

𑀢𑀻𑀭 𑀓𑀫𑁆𑀫 𑀲𑀫𑀢𑁆𑀢𑀺𑀫𑁆𑀳𑀺 𑀧𑀸𑀭 𑀲𑀸𑀫𑀢𑁆𑀣𑀺𑀬𑀸’𑀤𑀺𑀲𑀼;

𑁧𑁪𑁧.

𑀢𑀼𑀮𑀼’𑀫𑁆𑀫𑀸𑀦𑁂 𑀔𑀮 𑀲𑁄𑀯𑁂 𑀲𑀜𑁆𑀘𑀬𑁂 𑀧𑀸𑀮𑀭𑀓𑁆𑀔𑀡𑁂,

𑀓𑀮 𑀲𑀗𑁆𑀓𑀮𑀦𑀸’𑀤𑀻𑀲𑀼 (𑀪𑀯𑁂) 𑀫𑀻𑀮 𑀦𑀺𑀫𑀻𑀮𑀦𑁂;

𑁧𑁪𑁨.

𑀲𑀻𑀮𑀽’𑀧𑀥𑀸𑀭𑀡𑁂 𑀫𑀽𑀮 𑀭𑁄𑀳𑀡𑁂 𑀮𑀮 𑀇𑀘𑁆𑀙𑀦𑁂,

𑀤𑀼𑀮 𑀉𑀓𑁆𑀔𑁂𑀧𑀡𑁂 𑀧𑀽𑀮 𑀫𑀳𑀢𑁆𑀢𑀦 𑀲𑀫𑀼𑀲𑁆𑀲𑀬𑁂;

𑁧𑁪𑁩. ¶

𑀖𑀼𑀲 𑀲𑀤𑁆𑀤𑁂 𑀧𑀺𑀲 𑀧𑁂𑀲𑁂 𑀪𑀼𑀲𑀸’𑀮𑀗𑁆𑀓𑀭𑀡𑁂 (𑀲𑀺𑀬𑀸,)

𑀭𑀼𑀲 𑀧𑀸𑀭𑀼𑀲𑀺𑀬𑁂 𑀔𑀼𑀁𑀲 𑀅𑀓𑁆𑀓𑁄𑀲𑁂 𑀧𑀼𑀲 𑀧𑁄𑀲𑀦𑁂;

𑁧𑁪𑁪.

𑀤𑀺𑀲 𑀉𑀘𑁆𑀘𑀸𑀭𑀡𑀸’𑀤𑀻𑀲𑀼 𑀯𑀲 𑀅𑀘𑁆𑀙𑀸𑀤𑀦𑁂 (𑀲𑀺𑀬𑀸,)

𑀭𑀲’𑀲𑁆𑀲𑀸𑀤𑁂 𑀭𑀯𑁂 𑀲𑁆𑀦𑁂𑀳𑁂 (𑀅𑀣𑁄) 𑀲𑀺𑀲 𑀯𑀺𑀲𑁂𑀲𑀦𑁂;

𑁧𑁪𑁫.

𑀲𑀺 𑀩𑀦𑁆𑀥𑁂 𑀫𑀺𑀲𑁆𑀲 𑀲𑀫𑁆𑀫𑀺𑀲𑁆𑀲𑁂 𑀓𑀼𑀳 𑀯𑀺𑀫𑁆𑀪𑀸𑀧𑀦𑁂 𑀲𑀺𑀬𑀸,

𑀭𑀳 𑀘𑀸𑀕𑁂 𑀕𑀢𑁂 (𑀘𑀸’𑀧𑀺) 𑀫𑀳 𑀧𑀽𑀚𑀸𑀬 (𑀫𑀻𑀭𑀺𑀢𑁄;)

𑁧𑁪𑁬.

𑀧𑀺𑀳𑀺’𑀘𑁆𑀙𑀸𑀬𑀁 𑀲𑀺𑀬𑀸 𑀯𑀻𑀴 𑀮𑀚𑁆𑀚𑀸𑀬𑀁 𑀏𑀴 𑀨𑀸𑀴𑀦𑁂

𑀳𑀻𑀴 𑀕𑀸𑀭𑀳𑀺𑀬𑁂 𑀧𑀻𑀴 𑀩𑀸𑀥𑀸𑀬𑀁 𑀢𑀴 𑀢𑀸𑀴𑀦𑁂

𑀮𑀴 (𑀥𑀸𑀢𑀽)’𑀧𑀲𑁂𑀯𑀸’𑀬𑀁 (𑀯𑀢𑁆𑀢𑀢𑀻’𑀫𑁂𑀘𑀼𑀭𑀸𑀤𑀬𑁄;)

𑀲𑀫𑀢𑁆𑀢𑀸 𑀲𑀢𑁆𑀢𑀕𑀡𑀸.

𑁧𑁪𑁭. ¶

𑀪𑀼𑀯𑀸𑀤𑀻 𑀘 𑀭𑀼𑀥𑀸𑀤𑀻 𑀘-𑀤𑀺𑀯𑀸𑀤𑀺 𑀲𑁆𑀯𑀸’𑀤𑀬𑁄 𑀕𑀡𑀸,

𑀓𑀺𑀬𑀸𑀤𑀻 𑀘 𑀢𑀦𑀸𑀤𑀻 𑀘-𑀘𑀼𑀭𑀸𑀤𑀻𑀢𑀻’𑀥 𑀲𑀢𑁆𑀢𑀥𑀸;

𑁧𑁪𑁮.

𑀓𑀺𑀭𑀬𑀸𑀯𑀸𑀘𑀺𑀢𑁆𑀢𑀫𑀓𑁆𑀔𑀸𑀢𑀼-𑀫𑁂’𑀓𑁂𑀓𑀢𑁆𑀣𑁄 𑀩𑀳𑀽’𑀤𑀺𑀢𑁄,

𑀧𑀬𑁄𑀕𑀢𑁄’𑀦𑀼𑀕𑀦𑁆𑀢𑀩𑁆𑀩𑀸-𑀅𑀦𑁂𑀓𑀢𑁆𑀣𑀸 𑀳𑀺 𑀥𑀸𑀢𑀯𑁄;

𑁧𑁪𑁯.

𑀳𑀺𑀢𑀸𑀬 𑀫𑀦𑁆𑀤𑀩𑀼𑀤𑁆𑀥𑀻𑀦𑀁-𑀯𑁆𑀬𑀢𑁆𑀢𑀁 𑀯𑀡𑁆𑀡𑀓𑁆𑀓𑀫𑀸 𑀮𑀳𑀼𑀁,

𑀭𑀘𑀺𑀢𑀸 𑀥𑀸𑀢𑀼𑀫𑀜𑁆𑀚𑀼𑀲𑀸-𑀲𑀻𑀮𑀯𑀁𑀲𑁂𑀦 𑀥𑀻𑀫𑀢𑀸;

𑁧𑁫𑁦.

𑀲𑀤𑁆𑀥𑀫𑁆𑀫𑀧𑀗𑁆𑀓𑁂𑀭𑀼𑀳𑀭𑀸𑀚𑀳𑀁𑀲𑁄,

𑀆𑀲𑀺𑀝𑁆𑀞𑀥𑀫𑁆𑀫𑀝𑁆𑀞𑀺𑀢𑀺 𑀲𑀻𑀮𑀯𑀁𑀲𑁄;

𑀬𑀓𑁆𑀔𑀤𑁆𑀤𑀺𑀮𑁂𑀦𑀸𑀔𑁆𑀬 𑀦𑀺𑀯𑀸𑀲𑀯𑀸𑀲𑀻,

𑀬𑀢𑀺𑀲𑁆𑀲𑀭𑁄 𑀲𑁄𑀬𑀫𑀺𑀤𑀁 𑀅𑀓𑀸𑀲𑀺;

𑀓𑀘𑁆𑀘𑀸𑀬𑀦 𑀥𑀸𑀢𑀼𑀫𑀜𑁆𑀚𑀽𑀲𑀸 𑀲𑀫𑀢𑁆𑀢𑀸.

𑀲𑀸𑀘𑀭𑀺𑀬𑀸𑀦𑀼𑀲𑀺𑀝𑁆𑀞𑀸 ¶ 𑀧𑀭𑀺𑀲𑀺𑀝𑁆𑀞𑀧𑀭𑀺𑀪𑀸𑀲𑀸

𑁧.

𑀏𑀓𑀸 𑀦𑁂𑀓𑀲𑁆𑀲 𑀭𑀸𑀦𑀦𑁆𑀢𑀽-’𑀪𑀬𑁂𑀲𑀁 𑀅𑀦𑁆𑀢𑀺𑀫𑀸 𑀲𑀭𑀸,

𑀅𑀗𑁆𑀕𑀸𑀦𑀼𑀩𑀦𑁆𑀥𑀸 𑀥𑀸𑀢𑀽𑀦𑀁-𑀯𑀼𑀘𑁆𑀘𑀦𑁆𑀢𑁂’𑀧𑀺 𑀬𑀣𑀸𑀓𑁆𑀓𑀫𑀁;

𑁨.

𑀥𑀸𑀢𑀼𑀦𑁄 𑀯𑁆𑀬𑀸𑀜𑁆𑀚𑀦𑀸 𑀧𑀼𑀩𑁆𑀩𑁂-𑀦𑀺𑀕𑁆𑀕𑀳𑀻𑀢𑀁 𑀲𑀫’𑀦𑁆𑀢𑀺𑀫𑀸,

𑀇𑀯𑀡𑁆𑀡𑁂𑀦𑀸’𑀭𑀼𑀥𑀸𑀤𑀻𑀦-𑀫𑀦𑀼𑀩𑀦𑁆𑀥𑁂𑀦 𑀘𑀺𑀡𑁆𑀳𑀺𑀢𑀁;

𑁩.

𑀲𑁂𑀲𑀸’𑀦𑀼𑀩𑀦𑁆𑀥𑀸 𑀲𑀩𑁆𑀩𑁂𑀲𑀁-𑀳𑁄𑀦𑁆𑀢𑀻’𑀥𑀼’𑀘𑁆𑀘𑀸𑀭𑀡𑀧𑁆𑀨𑀮𑀸,

𑀉𑀘𑁆𑀘𑀸𑀯𑀘𑀧𑁆𑀨𑀮𑀸 𑀪𑀸𑀲-𑀦𑁆𑀢𑀭𑀫𑁆𑀧𑀢𑁆𑀯𑀸 𑀪 𑀯𑀦𑁆𑀢𑀺’𑀧𑀺;

𑁪. ¶

𑀦𑀸𑀫𑀥𑀸𑀢𑀼𑀓 𑀪𑀸𑀯𑁄’𑀧𑀺-𑀓𑀺𑀭𑀬𑀸𑀬 𑀅𑀥𑀺𑀓𑀸𑀭𑀢𑁄,

𑀯𑀺𑀭𑀼𑀤𑁆𑀥𑀦𑁆𑀢𑀭𑀸𑀪𑀸𑀯𑀸-𑀓𑁆𑀯𑀘𑀺𑀤𑁂𑀯 𑀧𑀬𑀼𑀚𑁆𑀚𑀢𑁂;

𑁫.

𑀤𑁆𑀯𑀦𑁆𑀤𑀬𑀼𑀢𑁆𑀢𑀺𑀯𑀲𑀸 𑀓𑁆𑀯𑀸𑀧𑀺-𑀆𑀤𑁂𑀲𑁄 𑀬𑁄𑀯𑀺𑀪𑀢𑁆𑀢𑀺𑀬𑀸,

𑀕𑀼𑀡𑀸𑀤𑀺𑀪𑀸𑀯 𑀲𑀤𑁆𑀤𑁄’𑀧𑀺-𑀢𑀓𑀺𑀭𑀬𑀢𑁆𑀣𑁂 𑀯𑀺𑀥𑀻𑀬𑀢𑁂;
