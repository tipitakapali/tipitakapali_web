𑀥𑀫𑁆𑀫𑀼’𑀚𑁆𑀚𑀮𑀦𑁆𑀢 𑀭𑀼𑀘𑀺𑀦𑁄, 𑀫𑀼𑀦𑀺𑀦𑁆𑀤𑁄’𑀤𑀸𑀢 𑀭𑁄𑀘𑀺𑀦𑁄.

𑀦𑀺𑀫𑀺𑀢𑁆𑀢

𑁨.

𑀧𑀺𑀗𑁆𑀕𑀮𑀸’𑀘𑀭𑀺𑀬𑀸𑀤𑀻𑀳𑀺, 𑀙𑀦𑁆𑀤𑀁 𑀬 𑀫𑀼𑀤𑀺𑀢𑀁 𑀧𑀼𑀭𑀸;

𑀕𑀦𑁆𑀣𑀧𑀭𑀺𑀫𑀸𑀡

𑁩.

𑀢𑀢𑁄 𑀫𑀸𑀕𑀥𑀪𑀸𑀲𑀸𑀬, 𑀫𑀢𑁆𑀢𑀸,𑀯𑀡𑁆𑀡,𑀯𑀺𑀪𑁂𑀤𑀦𑀁;

𑀮𑀓𑁆𑀔𑁆𑀬 𑀮𑀓𑁆𑀔𑀡 𑀲𑀁𑀬𑀼𑀢𑁆𑀢𑀁, 𑀧𑀲𑀦𑁆𑀦’𑀢𑁆𑀣,𑀧𑀤,𑀓𑁆𑀓𑀫𑀁.

𑀅𑀪𑀺𑀥𑀸𑀦𑀸𑀤𑀺

𑁪.

𑀆𑀭𑀪𑀺𑀲𑁆𑀲’𑀫𑀳𑀁 𑀤𑀸𑀦𑀺, 𑀢𑁂𑀲𑀁 𑀲𑀼𑀔𑀯𑀺𑀩𑀼𑀤𑁆𑀥𑀺𑀬𑀸.

𑀕𑀡𑀲𑀗𑁆𑀓𑁂𑀢𑀲𑀜𑁆𑀜𑀸

𑁫.

𑀲𑀩𑁆𑀩𑀕𑁆𑀮𑀸 𑀫𑁆𑀦𑀸,’𑀤𑀺𑀕𑀮𑀳𑀽, 𑀪𑁆𑀬𑀸’,𑀫𑀚𑁆𑀛’𑀦𑁆𑀢 𑀕𑀭𑀽 𑀚𑀲𑀸;

𑀫𑀚𑁆𑀛’𑀦𑁆𑀢𑀮𑀸 𑀭,𑀢𑁂’𑀢𑁂’𑀝𑁆𑀞, 𑀕𑀡𑀸 𑀕𑁄 𑀕𑀭𑀼,𑀮𑁄 𑀮𑀳𑀼.

𑀕𑀡𑀦𑀺𑀬𑀫

𑁬.

𑀪,𑀚,𑀲𑀸 ¶ 𑀲𑀩𑁆𑀩𑀕,𑀮𑀳𑀽, 𑀧𑀜𑁆𑀘𑀺’𑀫𑁂 𑀲𑀡𑁆𑀞𑀺𑀢𑀸 𑀕𑀡𑀸;

𑀅𑀭𑀺𑀬𑀸𑀤𑀺𑀫𑁆𑀳𑀺 𑀯𑀺𑀜𑁆𑀜𑁂𑀬𑁆𑀬𑀸, 𑀕𑀡𑁄 𑀇𑀥 𑀘𑀢𑀼’𑀓𑁆𑀓𑀮𑁄.

𑀕𑀭𑀼,𑀮𑀳𑀼𑀲𑀭𑀽𑀧

𑁭.

𑀲𑀁𑀬𑁄𑀕𑀸’𑀤𑀺 𑀘, 𑀤𑀻𑀖𑁄 𑀘, 𑀦𑀺𑀕𑁆𑀕𑀳𑀻𑀢𑀧𑀭𑁄 𑀘, 𑀬𑁄;

𑀕𑀭𑀼, 𑀯𑀗𑁆𑀓𑁄, 𑀧𑀸𑀤𑀦𑁆𑀢𑁄,𑀯𑀸, 𑀭𑀲𑁆𑀲𑁄’𑀜𑁆𑀜𑁄 𑀫𑀢𑁆𑀢𑀺𑀓𑁄 𑀮𑀼’𑀚𑀼.

𑁮.

𑀧𑀭𑁂 𑀧𑀸𑀤𑀸𑀤𑀺𑀲𑀁𑀬𑁄𑀕𑁂, 𑀬𑁄 𑀧𑀼𑀩𑁆𑀩𑁄 𑀕𑀭𑀼𑀓’𑀓𑁆𑀔𑀭𑁄;

𑀮𑀳𑀼 𑀲 𑀓𑁆𑀯𑀘𑀺 𑀯𑀺𑀜𑁆𑀜𑁂𑀬𑁆𑀬𑁄, 𑀢𑀤𑀼𑀤𑀸𑀳𑀭𑀡𑀁 𑀬𑀣𑀸.

𑁯.

𑀤𑀲𑁆𑀲𑀦𑀭𑀲𑀸’𑀦𑀼𑀪𑀸𑀯𑀦𑁂, 𑀦𑀺𑀩𑀤𑁆𑀥𑀕𑁂𑀥𑀸 𑀚𑀺𑀦𑀲𑁆𑀲’𑀬𑀁 𑀚𑀦𑀢𑀸;

𑀯𑀺𑀫𑁆𑀳𑀬𑀚𑀦𑀦𑀻 𑀲𑀜𑁆𑀜𑀢, 𑀓𑁆𑀭𑀺𑀬𑀸 𑀦𑀼 𑀓𑀁 𑀦𑀸’𑀦𑀼𑀭𑀜𑁆𑀚𑀬𑀢𑀺.

𑁧𑁦.

𑀯𑀺𑀜𑁆𑀜𑁂𑀬𑁆𑀬𑀸 𑀮𑁄𑀓𑀢𑁄 𑀲𑀜𑁆𑀜𑀸, 𑀲𑀫𑀼𑀤𑁆𑀤𑁄,𑀲𑀼,𑀭𑀲𑀸𑀤𑀺𑀦𑀁;

𑀧𑀸𑀤𑁄𑀜𑁂𑀬𑁆𑀬𑁄 𑀘𑀢𑀼𑀢𑁆𑀣𑀁’𑀲𑁄, 𑀧𑀤𑀘𑁆𑀙𑁂𑀤𑁄 𑀬𑀢𑀻 𑀪𑀯𑁂.

𑁧𑁧.

𑀲𑀫,𑀫𑀟𑁆𑀠𑀲𑀫𑀁, 𑀯𑀼𑀢𑁆𑀢𑀁, 𑀯𑀺𑀲𑀫𑀁 𑀘𑀸’𑀧𑀭𑀁 𑀢𑀺𑀥𑀸;

𑀲𑀫𑀸 𑀮𑀓𑁆𑀔𑀡𑀢𑁄 𑀧𑀸𑀤𑀸, 𑀘𑀢𑁆𑀢𑀸𑀭𑁄 𑀬𑀲𑁆𑀲 𑀢𑀁 𑀲𑀫𑀁.

𑁧𑁨.

𑀬𑀲𑁆𑀲’𑀦𑁆𑀢𑀺𑀫𑁂𑀦 𑀤𑀼𑀢𑀺𑀬𑁄, 𑀢𑀢𑀺𑀬𑁂𑀦𑀸’𑀤𑀺𑀫𑁄 𑀲𑀫𑁄;

𑀢’𑀤𑀟𑁆𑀠𑀲𑀫, 𑀫𑀜𑁆𑀜𑀁 𑀢𑀼, 𑀪𑀺𑀦𑁆𑀦 𑀮𑀓𑁆𑀔𑀡 𑀧𑀸𑀤𑀺𑀓𑀁.

𑁧𑁩.

𑀧𑀸𑀤’𑀫𑁂𑀓𑀓𑁆𑀔𑀭𑀸’𑀭𑀩𑁆𑀪, 𑀬𑀸𑀯 𑀙𑀩𑁆𑀩𑀻𑀲𑀢’𑀓𑁆𑀔𑀭𑀸;

𑀪𑀯𑁂 𑀧𑀸𑀤𑁂𑀳𑀺 𑀢𑀁 𑀙𑀦𑁆𑀤𑀁, 𑀦𑀸𑀦𑀸𑀦𑀸𑀫𑁄’𑀤𑀺𑀢𑀁 𑀢𑀢𑁄.

𑁧𑁪.

𑀤𑀡𑁆𑀟𑀓𑀸 𑀘𑀡𑁆𑀟𑀯𑀼𑀝𑁆𑀞𑁆𑀬𑀸’𑀤𑀺, 𑀧𑀸𑀤𑁂𑀳𑀺 𑀙𑀳𑀺, 𑀢𑀻𑀳𑀺 𑀢𑀼;

‘𑀕𑀸𑀣𑀸’𑀢𑀺 𑀘 𑀧𑀭𑀢𑁆𑀣𑁂’𑀯𑀁, 𑀙𑀦𑁆𑀤𑁄 𑀲𑀜𑁆𑀜𑀸 𑀧𑀓𑀸𑀲𑀺𑀢𑀸.

𑁧𑁫.

𑀅𑀦𑀦𑁆𑀢𑀭𑁄’𑀤𑀺𑀢𑀁 𑀘’𑀜𑁆𑀜, 𑀫𑁂𑀢𑀁 𑀲𑀸𑀫𑀜𑁆𑀜 𑀦𑀸𑀫𑀢𑁄;

‘𑀕𑀸𑀣𑀸’𑀇𑀘𑁆𑀘𑁂𑀯 𑀦𑀺𑀤𑁆𑀤𑀺𑀝𑁆𑀞𑀁, 𑀫𑀼𑀦𑀺𑀦𑁆𑀤𑀯𑀘𑀦𑁂 𑀧𑀦.

𑁧𑁬.

𑀯𑀺𑀲𑁂𑀲𑀦𑀸𑀫𑀢𑁄 𑀓𑀺𑀜𑁆𑀘𑀺, 𑀕𑀳𑁂𑀢𑁆𑀯𑀸 𑀲𑀩𑁆𑀩𑀣𑁄’𑀘𑀺𑀢𑀁;

𑀤𑀲𑁆𑀲𑀬𑀺𑀲𑁆𑀲𑀸𑀫’𑀳𑀁 𑀢𑁂’𑀢𑁆𑀣, 𑀦𑀸𑀫𑀸𑀦𑀸’𑀯𑀺 𑀪𑀯𑀺𑀲𑁆𑀲𑀭𑁂.

𑀇𑀢𑀺 𑀯𑀼𑀢𑁆𑀢𑁄𑀤𑀬𑁂 𑀙𑀦𑁆𑀤𑀲𑀺 𑀲𑀜𑁆𑀜𑀸𑀧𑀭𑀺𑀪𑀸𑀲𑀸 𑀦𑀺𑀤𑁆𑀤𑁂𑀲𑁄 𑀦𑀸𑀫

𑀧𑀞𑀫𑁄 𑀧𑀭𑀺𑀘𑁆𑀙𑁂𑀤𑁄.

𑁨. 𑀫𑀢𑁆𑀢𑀸𑀯𑀼𑀢𑁆𑀢𑀺𑀦𑀺𑀤𑁆𑀤𑁂𑀲-𑀤𑀼𑀢𑀺𑀬𑀧𑀭𑀺𑀘𑁆𑀙𑁂𑀤

𑀕𑀡𑀦𑀺𑀬𑀫

𑁧𑁭.

𑀙𑀝𑁆𑀞𑁄’𑀔𑀺𑀮𑀮𑀳𑀼,𑀚𑁄 ¶ 𑀯𑀸,

𑀕𑀬𑀼𑀢𑀸’𑀜𑁆𑀜𑁂,𑀙’𑀕𑁆𑀕𑀡𑀸,𑀦 𑀚𑁄 𑀯𑀺𑀲𑀫𑁂,;

𑀅𑀭𑀺𑀬𑀸𑀬’𑀦𑁆𑀢𑀟𑁆𑀠𑁂 𑀮𑁄, 𑀙𑀝𑁆𑀞𑁄,’𑀦𑁆𑀢𑁂 𑀕𑁄,𑀕𑀡𑀸 𑀙’𑀜𑁆𑀜𑁂.

𑀬𑀢𑀺𑀦𑀺𑀬𑀫

𑁧𑁮.

𑀧𑀞𑀫𑀟𑁆𑀠𑁂 𑀙𑀝𑁆𑀞𑁄 𑀘𑁂,

𑀲𑀩𑁆𑀩𑀮𑀳𑁂,’𑀢𑁆𑀣𑀸’𑀤𑀺𑀮𑀳𑀼𑀦𑀺 𑀪𑀯𑀢𑀺𑀬𑀢𑀺;

𑀢𑀧𑁆𑀧𑀭𑀓𑁄,𑀦𑁆𑀢𑁂𑀧𑀺, 𑀲𑀘𑁂, 𑀘𑀭𑀺𑀫𑁂𑀧𑀺, 𑀪𑀯𑀢𑀺 𑀘𑀢𑀼𑀢𑁆𑀣𑁄’𑀦𑁆𑀢𑁂.

𑁧𑁯. 𑀅𑀭𑀺𑀬𑀸𑀲𑀸𑀫𑀜𑁆𑀜𑀁 𑀘𑁂, 𑀧𑀼𑀩𑁆𑀩𑁄’𑀤𑀺𑀢 𑀮𑀓𑁆𑀔𑀡𑀁 𑀪𑀯𑁂 𑀬𑀲𑁆𑀲𑀸.

𑁨𑁦. 𑀆𑀤𑀺𑀫’𑀫𑀣 𑀧𑀸𑀤𑀬𑀼𑀕𑀁, 𑀬𑀲𑁆𑀲𑀸 𑀢𑁆𑀬𑀁’𑀲𑁂𑀳𑀺 𑀲𑀸 𑀧𑀣𑁆𑀬𑀸.

𑁨𑁧.

𑀬𑀢𑁆𑀣 𑀕𑀡𑀢𑁆𑀢𑀬 𑀫𑀼𑀮𑁆𑀮𑀗𑁆𑀖𑀺,

𑀬𑁄’𑀪𑀬𑀢𑁆𑀣𑀸’𑀤𑀺𑀫𑁄 𑀪𑀯𑁂 𑀯𑀺𑀧𑀼𑀮𑀸.

𑁨𑁨. 𑀕𑀭𑀼𑀫𑀚𑁆𑀛𑀕𑁄 𑀚𑀓𑀸𑀭𑁄, 𑀘𑀢𑀼𑀢𑁆𑀣𑀓𑁄 𑀤𑀼𑀢𑀺𑀬𑀓𑁄 𑀘𑀧𑀮𑀸.

𑁨𑁩.

𑀘𑀧𑀮𑀸’𑀕𑀢𑀸’𑀔𑀺𑀮𑀁 𑀘𑁂, 𑀤𑀮𑀸’𑀤𑀺𑀫𑀁 𑀮𑀓𑁆𑀔𑀡𑀁 𑀪𑀚𑀢𑀺 𑀬𑀲𑁆𑀲𑀸;

𑀧𑀣𑁆𑀬𑀸𑀮𑀓𑁆𑀔𑀡’𑀫𑀜𑁆𑀜𑀁, 𑀫𑀼𑀔𑀘𑀧𑀮𑀸 𑀦𑀸𑀫 𑀲𑀸 𑀪𑀯𑀢𑀺.

𑁨𑁪.

𑀧𑀣𑁆𑀬𑀸𑀬 𑀮𑀓𑁆𑀔𑀡𑀁 𑀘𑁂, 𑀧𑀞𑀫𑀟𑁆𑀠𑁂 𑀮𑀓𑁆𑀔𑀡𑀁 𑀢𑀼 𑀘𑀧𑀮𑀸𑀬;

𑀤𑀼𑀢𑀺𑀬𑁂 𑀤𑀮𑁂’𑀣 𑀬𑀲𑁆𑀲𑀸, 𑀧𑀓𑀺𑀢𑁆𑀢𑀺𑀢𑀸 𑀲𑀸 𑀚𑀖𑀦𑀘𑀧𑀮𑀸.

𑀅𑀭𑀺𑀬𑀸𑀚𑀸𑀢𑀺𑀬𑁄.

𑁨𑁫.

𑀲𑀩𑁆𑀩𑀁𑀧𑀞𑀫𑀤𑀮𑁂 𑀬𑀤𑀺, 𑀮𑀓𑁆𑀔𑀡’𑀫𑀭𑀺𑀬𑀸𑀬 𑀯𑀼𑀢𑁆𑀢’𑀫𑀼𑀪𑀬𑁂𑀲𑀼;

𑀬𑀲𑁆𑀲𑀸 𑀤𑀮𑁂𑀲𑀼 𑀬𑀼𑀢𑁆𑀢𑀁,

𑀯𑀼𑀢𑁆𑀢𑀸 𑀲𑀸 𑀕𑀻𑀢𑀺 𑀯𑀼𑀢𑁆𑀢 𑀬𑀢𑀺 𑀮𑀮𑀺𑀢𑀸.

𑁨𑁬.

𑀅𑀭𑀺𑀬𑀸𑀬𑀁 𑀤𑀼𑀢𑀺𑀬’𑀟𑁆𑀠𑁂, 𑀕𑀤𑀺𑀢𑀸’𑀔𑀺𑀮𑀮𑀓𑁆𑀔𑀡𑀁 𑀬𑀁 𑀢𑀁;

𑀪𑀯𑀢𑀺 𑀤𑀮𑁂𑀲𑀼’𑀪𑀬𑁂𑀲𑀼𑀧𑀺,

𑀬𑀤𑀺 𑀬𑀲𑁆𑀲𑀸 𑀲𑀸’𑀬 𑀫𑀼𑀧𑀕𑀻𑀢𑀺.

𑁨𑁭.

𑀅𑀭𑀺𑀬𑀸𑀬’𑀟𑁆𑀠𑀤𑁆𑀯𑀺𑀢𑀬𑀁, 𑀧𑀼𑀩𑁆𑀩𑁄𑀤𑀺𑀢 𑀮𑀓𑁆𑀔𑀡𑁄’𑀧𑁂𑀢𑀁;

𑀯𑀺𑀧𑀭𑀺𑀬𑀬𑁂𑀦𑀸’𑀪𑀺𑀳𑀺𑀢𑀁,

𑀬𑀲𑁆𑀲𑀸 𑀲𑀫𑁆𑀪𑀯𑀢𑀺 𑀘𑁂’𑀳 𑀲𑁄’𑀕𑁆𑀕𑀻𑀢𑀺.

𑁨𑁮.

𑀅𑀭𑀺𑀬𑀸𑀧𑀼𑀩𑁆𑀩’𑀟𑁆𑀠𑀁 ¶ 𑀬𑀤𑀺, 𑀕𑀭𑀼𑀦𑁂’𑀓𑁂𑀦𑀸’𑀥𑀺𑀓𑁂𑀦 𑀦𑀺𑀥𑀦𑁂 𑀬𑀼𑀢𑁆𑀢𑀁;

𑀬𑀤𑀺 𑀧𑀼𑀩𑁆𑀩’𑀟𑁆𑀠𑀲𑀫𑀸𑀦𑀁, 𑀤𑀮 𑀫𑀺𑀢𑀭𑀁 𑀘𑁄’𑀤𑀺𑀢𑀸’𑀬’𑀫𑀭𑀺𑀬𑀸𑀕𑀻𑀢𑀺.

𑀕𑀻𑀢𑀺𑀚𑀸𑀢𑀺𑀬𑁄.

𑁨𑁯.

𑀯𑀺𑀲𑀫𑁂 𑀙 𑀲𑀺𑀬𑀼𑀁 𑀓𑀮𑀸 𑀫𑀼𑀔𑁂,

𑀲𑀫𑁂 𑀢𑁆𑀯’𑀝𑁆𑀞, 𑀭,𑀮,𑀕𑀸, 𑀢𑀢𑁄’𑀧𑀭𑀺;

𑀯𑁂𑀢𑀸𑀮𑀻𑀬𑀁 𑀢 𑀫𑀼𑀘𑁆𑀘𑀢𑁂, 𑀮𑀳𑀼 𑀙𑀓𑁆𑀓𑀁 𑀦 𑀦𑀺𑀭𑀦𑁆𑀢𑀭𑀁 𑀲𑀫𑁂.

𑁩𑁦.

𑀯𑁂𑀢𑀸𑀮𑀻𑀬𑁄𑀧𑀫𑀁 𑀫𑀼𑀔𑁂 𑀢𑀁,

𑀑𑀧𑀘𑁆𑀙𑀦𑁆𑀤𑀲𑀓𑀁 𑀭,𑀬𑀸 𑀬’𑀤𑀦𑁆𑀢𑁂.

𑁩𑁧. 𑀆𑀧𑀸𑀢𑀮𑀺𑀓𑀸 𑀓𑀣𑀺𑀢𑀸’𑀬𑀁, 𑀪𑀕𑀕𑀸’𑀦𑁆𑀢𑁂 𑀬𑀤𑀺 𑀧𑀼𑀩𑁆𑀩𑀫𑀺𑀯’𑀜𑁆𑀜𑀁.

𑁩𑁨.

𑀬𑀤𑀸’𑀤𑀺𑀢𑁄 𑀤𑀓𑁆𑀔𑀺𑀡𑀦𑁆𑀢𑀺𑀓𑀸,

𑀞𑀺𑀢𑁂’𑀢𑁆𑀣 𑀧𑀸𑀤𑁂𑀲𑁆𑀯𑀸’𑀔𑀺𑀮𑁂𑀲𑀼 𑀚𑁄.

𑁩𑁩.

‘𑀉𑀤𑀺𑀘𑁆𑀘𑀯𑀼𑀢𑁆𑀢𑀻’𑀢𑀺 𑀯𑀼𑀘𑁆𑀘𑀢𑁂,

𑀚𑁄 𑀘𑀸’𑀤𑁄 𑀯𑀺𑀲𑀫𑁂𑀲𑀼 𑀲𑀡𑁆𑀞𑀺𑀢𑁄.

𑁩𑁪. 𑀧𑀼𑀩𑁆𑀩𑀢𑁆𑀣, 𑀲𑀫𑁂𑀲𑀼 𑀘𑁂 𑀕, 𑀚𑀸, ‘𑀧𑀘𑁆𑀘𑀯𑀼𑀢𑁆𑀢𑀺’ 𑀭𑀼𑀤𑀺𑀢𑀸’𑀢𑀺 𑀲𑀡𑁆𑀞𑀺𑀢𑀸.

𑁩𑁫.

𑀲𑀫𑀸𑀲𑀫𑀸’𑀢𑁆𑀭𑀸’𑀤𑀺𑀦𑀁 𑀲𑀫𑀸,

𑀲𑀁𑀬𑀼𑀢𑀸 𑀪𑀯𑀢𑀺 𑀢𑀁 𑀧𑀯𑀢𑁆𑀢𑀓𑀁.

𑁩𑁬. 𑀅𑀲𑁆𑀲 𑀲𑀸 𑀲𑀫 𑀓𑀢𑀸’ 𑀧𑀭𑀦𑁆𑀢𑀺𑀓𑀸.

𑁩𑁭. 𑀢𑀤’𑀜𑁆𑀜𑀚𑀸 𑀘𑀸𑀭𑀼𑀳𑀸𑀲𑀺𑀦𑀻.

𑀯𑁂𑀢𑀸𑀮𑀻𑀬𑀚𑀸𑀢𑀺𑀬𑁄.

𑁩𑁮. 𑀤𑁆𑀯𑀺𑀓 𑀯𑀺𑀳𑀢 𑀯𑀲𑀼 𑀮𑀳𑀼 𑀅𑀘𑀮𑀥𑀺𑀢𑀺 𑀭𑀺’𑀳.

𑁩𑁯. 𑀫𑀢𑁆𑀢𑀸𑀲𑀫𑀓𑀁 𑀦𑀯𑀫𑁄 𑀮𑁆𑀕’𑀦𑁆𑀢𑁂.

𑁪𑁦. 𑀚𑁄 𑀦𑁆𑀮𑀸’ 𑀣𑀯𑀸’𑀡𑁆𑀡𑀯𑀸 𑀯𑀺𑀲𑀺𑀮𑁄𑀓𑁄.

𑁪𑁧. 𑀢𑀤𑁆𑀯𑀬𑀢𑁄 𑀯𑀸𑀦𑀯𑀸𑀲𑀺𑀓𑀸’𑀔𑁆𑀬𑀸.

𑁪𑁨. 𑀧𑀜𑁆𑀘,𑀝𑁆𑀞,𑀦𑀯𑀲𑀼 𑀬𑀤𑀺 𑀮𑁄 𑀘𑀺𑀢𑁆𑀭𑀸.

𑁪𑁩. 𑀕,𑀮𑁆𑀬𑀸’𑀝𑁆𑀞𑀳𑀺 𑀘𑁂’𑀲𑀸 𑀯𑀼’𑀧𑀘𑀺𑀢𑁆𑀭𑀸.

𑁪𑁪.

𑀬’𑀫𑀢𑀻𑀢 𑀮𑀓𑁆𑀔𑀡 𑀯𑀺𑀲𑁂𑀲 𑀬𑀼𑀢𑀁, (𑀘𑀺𑀢𑁆𑀭𑀸)

𑀫𑀢𑁆𑀢𑀸 𑀲𑀫𑀸’𑀤𑀺 𑀧𑀸𑀤𑀸’𑀪𑀺𑀳𑀺𑀢𑀁; (𑀯𑀺𑀲𑀺𑀮𑁄𑀓)

𑀅𑀦𑀺𑀬𑀢 𑀯𑀼𑀢𑁆𑀢 𑀧𑀭𑀺𑀫𑀸𑀡 𑀲𑀳𑀺𑀢𑀁, (𑀯𑀸𑀦𑀯𑀸𑀲𑀺𑀓𑀸)

𑀧𑀣𑀺𑀢𑀁 𑀚𑀦𑁂𑀲𑀼 𑀧𑀸𑀤𑀸𑀓𑀼𑀮𑀓𑀁. (𑀯𑀺𑀲𑀺𑀮𑁄𑀓)

𑀫𑀢𑁆𑀢𑀸𑀲𑀫𑀓 𑀚𑀸𑀢𑀺𑀬𑁄.

𑁪𑁫.

𑀯𑀺𑀦𑀸 ¶ 𑀯𑀡𑁆𑀡𑁂𑀳𑀺 𑀫𑀢𑁆𑀢𑀸 𑀕𑀸, 𑀯𑀺𑀦𑀸 𑀯𑀡𑁆𑀡𑀸 𑀕𑀭𑀽𑀳𑀺 𑀢𑀼;

𑀯𑀺𑀦𑀸 𑀮𑀳𑀽𑀳𑀺 𑀕𑀭𑀯𑁄, 𑀤𑀮𑁂 𑀧𑀣𑁆𑀬𑀸𑀤𑀺𑀦𑁄 𑀫𑀢𑀸.

𑀇𑀢𑀺 𑀯𑀼𑀢𑁆𑀢𑁄𑀤𑀬𑁂 𑀙𑀦𑁆𑀤𑀲𑀺 𑀫𑀢𑁆𑀢𑀸𑀯𑀼𑀢𑁆𑀢𑀺𑀦𑀺𑀤𑁆𑀤𑁂𑀲𑁄 𑀦𑀸𑀫

𑀤𑀼𑀢𑀺𑀬𑁄 𑀧𑀭𑀺𑀘𑁆𑀙𑁂𑀤𑁄.

𑁩. 𑀲𑀫𑀯𑀼𑀢𑁆𑀢𑀺𑀦𑀺𑀤𑁆𑀤𑁂𑀲-𑀢𑀢𑀺𑀬𑀧𑀭𑀺𑀘𑁆𑀙𑁂𑀤

𑁪𑁬. 𑀢𑁆𑀬𑀸’ 𑀘𑁂 𑀢𑀦𑀼𑀫𑀚𑁆𑀛𑀸.

𑀕𑀸𑀬𑀢𑁆𑀢𑀻.

𑁪𑁭. 𑀓𑀼𑀫𑀸𑀭 𑀮𑀮𑀺𑀢𑀸 𑀚𑁆𑀲𑁆𑀕𑀸.

𑀉𑀡𑁆𑀳𑀺𑀓𑀸.

𑁪𑁮. 𑀘𑀺𑀢𑁆𑀭𑀧𑀤𑀸 𑀬𑀤𑀺 𑀪𑀸 𑀕𑀸.

𑁪𑁯. 𑀫𑁄 𑀫𑁄 𑀕𑁄 𑀕𑁄 𑀯𑀺𑀚𑁆𑀚𑀼𑀫𑁆𑀫𑀸𑀮𑀸.

𑁫𑁦. 𑀪𑀸 𑀢, 𑀮, 𑀕𑀸 𑀫𑀸𑀡𑀯𑀓𑀁.

𑁫𑁧. 𑀕𑁆𑀮𑀸 𑀲𑀫𑀸𑀦𑀺𑀓𑀸 𑀭,𑀚𑀸 𑀘.

𑁫𑁨. 𑀧𑀫𑀸𑀡𑀺𑀓𑀸 𑀚, 𑀭𑀸 𑀮, 𑀕𑀸.

𑀅𑀦𑀼𑀝𑁆𑀞𑀼𑀪𑀸.

𑁫𑁩. 𑀭𑀸 𑀦,𑀲𑀸 𑀬𑀤𑀺 𑀳𑀮𑀫𑀼𑀔𑀻.

𑁫𑁪. 𑀪𑀼𑀚𑀕 𑀲𑀼𑀲𑀼 𑀲𑀝𑀸 𑀦𑀸 𑀫𑁄.

𑀩𑁆𑀭𑀳𑀢𑀻.

𑁫𑁫. 𑀫𑁆𑀲𑀸𑀚𑁆𑀕𑀸 𑀲𑀼𑀤𑁆𑀥𑀯𑀺𑀭𑀸𑀚𑀺𑀢𑀁 𑀫𑀢𑀁.

𑁫𑁬. 𑀫𑁆𑀦𑀸 𑀬𑁄 𑀕𑁄 𑀬𑀤𑀺 𑀧𑀡𑀯𑁄 𑀔𑁆𑀬𑀸𑀢𑁄.

𑁫𑁭. 𑀫𑁆𑀪𑀸 𑀲,𑀕𑀬𑀼𑀢𑁆𑀢𑀸 𑀭𑀼𑀫𑁆𑀫𑀯𑀢𑀻 𑀲𑀸.

𑁫𑁮. 𑀜𑁂𑀬𑁆𑀬𑀸 𑀫𑀢𑁆𑀢𑀸 𑀫, 𑀪, 𑀲, 𑀕𑀬𑀼𑀢𑁆𑀢𑀸.

𑁫𑁯. 𑀘𑀫𑁆𑀧𑀓𑀫𑀸𑀮𑀸 𑀘𑁂 𑀪, 𑀫, 𑀲𑀸 𑀕𑁄 [𑀇𑀤𑀁 𑀦𑀸𑀫𑀦𑁆𑀢𑀭𑀜𑀸𑀧𑀦𑀢𑁆𑀣𑀫𑁂𑀯 𑀧𑀼𑀦 𑀯𑀼𑀢𑁆𑀢𑀁 (𑀝𑀻.)].

𑁬𑁦. 𑀦, 𑀭, 𑀚, 𑀕𑁂𑀳𑀺 𑀲𑀸 𑀫𑀦𑁄𑀭𑀫𑀸.

𑁬𑁧. 𑀉𑀩𑁆𑀪𑀸𑀲𑀓𑀁 𑀢𑀁 𑀘𑁂 𑀢𑁄 𑀫, 𑀭𑀸 𑀮𑁆𑀘.

𑁬𑁨. 𑀢𑁄 𑀚𑀸 𑀕𑀭𑀼𑀦𑀸’𑀬’𑀫𑀼𑀧𑀝𑁆𑀞𑀺𑀢𑀸.

𑀧𑀦𑁆𑀢𑀺.

𑁬𑁩. 𑀇𑀦𑁆𑀤𑀸𑀤𑀺𑀓𑀸 ¶ 𑀢𑀸 𑀯𑀚𑀺𑀭𑀸 𑀚, 𑀕𑀸 𑀕𑁄.

𑁬𑁪. 𑀉𑀧𑀸𑀤𑀺𑀓𑀸 𑀲𑀸’𑀯 𑀚,𑀢𑀸 𑀚,𑀕𑀸 𑀕𑁄.

𑁬𑁫.

𑀅𑀦𑀦𑁆𑀢𑀭𑁄’𑀤𑀻𑀭𑀺𑀢 𑀮𑀓𑁆𑀔𑀡𑀸 𑀘𑁂, (𑀉𑀧𑁂𑀦𑁆𑀤𑀯𑀚𑀺𑀭)
