𑀮𑀓𑁆𑀔𑀡𑀸𑀢𑁄

𑀩𑀼𑀤𑁆𑀥𑀣𑁄𑀫𑀦𑀸𑀕𑀸𑀣𑀸

𑀩𑀼𑀤𑁆𑀥𑀯𑀦𑁆𑀤𑀦𑀸

𑀉𑀡𑁆𑀡𑀸𑀮𑁄𑀫𑀺𑀓𑀦𑀸𑀣 𑀯𑀦𑁆𑀤𑀦𑀸

𑀦𑀫𑁄 𑀢𑀲𑁆𑀲 𑀪𑀕𑀯𑀢𑁄 𑀅𑀭𑀳𑀢𑁄 𑀲𑀫𑁆𑀫𑀸𑀲𑀫𑁆𑀩𑀼𑀤𑁆𑀥𑀲𑁆𑀲

𑀮𑀓𑁆𑀔𑀡𑀸𑀢𑁄

𑀩𑀼𑀤𑁆𑀥𑀣𑁄𑀫𑀦𑀸𑀕𑀸𑀣𑀸

𑁧.

𑀪𑀯𑀸𑀪𑀯𑁂𑀲𑀼 ¶ 𑀦𑁂𑀓𑁂𑀲𑀼,

𑀧𑀽𑀭𑁂𑀲𑀺 𑀢𑀺𑀁𑀲𑀧𑀸𑀭𑀫𑀻;

𑀘𑀭𑀺𑀬𑀸𑀬𑁄𑀘 𑀲𑀫𑁆𑀩𑀼𑀤𑁆𑀥𑁄,

𑀧𑀭𑀺𑀘𑁆𑀘𑀸𑀕𑁂𑀚𑀳𑀁𑀲𑀤𑀸.

𑁨.

𑀯𑁂𑀲𑁆𑀲𑀦𑁆𑀢𑀭𑀢𑁆𑀢𑀪𑀸𑀯𑀫𑁆𑀳𑀺 ¶ ,

𑀞𑀺𑀢𑁄𑀫𑀤𑁆𑀤𑀺𑀁 𑀧𑀺𑀲𑁄𑀅𑀤𑀸;

𑀦𑀺𑀚𑀁𑀓𑀡𑁆𑀳𑀸𑀚𑀺𑀦𑀁𑀚𑀸𑀮𑀺𑀁,

𑀯𑀺𑀬𑀁𑀯𑀗𑁆𑀓𑀢𑀧𑀩𑁆𑀩𑀢𑁂.

𑁩.

𑀢𑀢𑁄𑀘𑀯𑀺𑀢𑁆𑀯𑀸𑀢𑀼𑀲𑀺𑀢𑁂,

𑀧𑀼𑀭𑁂𑀉𑀧𑁆𑀧𑀚𑁆𑀚𑀺𑀧𑀼𑀜𑁆𑀜𑀯𑀸;

𑀭𑀽𑀧𑀸𑀤𑀺𑀤𑀲𑀞𑀸𑀦𑁂𑀳𑀺 ¶ ,

𑀅𑀜𑁆𑀜𑁂𑀤𑁂𑀯𑁂𑀅𑀢𑀺𑀓𑁆𑀓𑀫𑀺.

𑁪.

𑀘𑀓𑁆𑀓𑀯𑀸𑀴𑀲𑀳𑀲𑁆𑀲𑁂𑀳𑀺,

𑀤𑀲𑀳𑀸𑀕𑀫𑁆𑀫𑀲𑀩𑁆𑀩𑀲𑁄;

𑀬𑀸𑀘𑀺𑀢𑁄𑀤𑁂𑀯𑀩𑁆𑀭𑀳𑁆𑀫𑀽𑀳𑀺,

𑀩𑀼𑀤𑁆𑀥𑀪𑀸𑀯𑀫𑀦𑁄𑀦𑀢𑁄.

𑁫.

𑀯𑀺𑀮𑁄𑀓𑀦𑀸𑀦𑀺𑀲𑁄𑀧𑀜𑁆𑀘 ¶ ,

𑀯𑀺𑀮𑁄𑀓𑁂𑀢𑁆𑀯𑀸𑀲𑀺𑀭𑀺𑀦𑁆𑀥𑀭𑁄;

𑀢𑀢𑁄𑀘𑀯𑀺𑀢𑁆𑀯𑀸𑀫𑀸𑀬𑀸𑀬,

𑀉𑀧𑁆𑀧𑀚𑁆𑀚𑀺𑀲𑀸𑀓𑀺𑀬𑁂𑀓𑀼𑀮𑁂.

𑁬.

𑀯𑀺𑀲𑀸𑀔𑀧𑀼𑀡𑁆𑀡𑀫𑁂𑀲𑁂𑀝𑁆𑀞𑁂 ¶ ,

𑀯𑀺𑀚𑀸𑀬𑀺𑀮𑀼𑀫𑁆𑀩𑀺𑀦𑀻𑀯𑀦𑁂;

𑀲𑀻𑀳𑀦𑀸𑀤𑀁𑀧𑀦𑀸𑀤𑁂𑀦𑁆𑀢𑁄,

𑀲𑀢𑁆𑀢𑀧𑀸𑀤𑀁𑀕𑀢𑁄𑀢𑀤𑀸.

𑁮.

𑀯𑀺𑀭𑁄𑀘𑀺𑀁𑀲𑀼𑀢𑀤𑀸𑀓𑀸𑀲𑁂 ¶ ,

𑀢𑀸𑀭𑁄𑀘𑀦𑁆𑀤𑀤𑀺𑀯𑀸𑀓𑀭𑀸;

𑀭𑀢𑀦𑀸𑀦𑀺𑀘𑀪𑀽𑀫𑀝𑁆𑀞𑀸,

𑀦𑀸𑀦𑀸𑀧𑀼𑀧𑁆𑀨𑀸𑀘𑀧𑀼𑀧𑁆𑀨𑀭𑁂.

𑁮.

𑀲𑀻𑀢𑀸𑀬𑀦𑁆𑀢𑀺 𑀅𑀯𑀻𑀘𑀕𑁆𑀕𑀻 ¶ ,

𑀔𑀼𑀧𑁆𑀧𑀺𑀧𑀸𑀲𑀸𑀘𑀲𑁄𑀳𑀺𑀢𑀸;

𑀮𑁄𑀓𑀦𑁆𑀢𑀭𑁂𑀫𑀳𑀸𑀧𑀪𑀸,

𑀅𑀦𑁆𑀥𑀸𑀧𑀲𑁆𑀲𑀦𑁆𑀢𑀺𑀢𑀸𑀯𑀤𑁂.

𑁯.

𑀅𑀖𑀝𑁆𑀝𑀺𑀢𑀸𑀯𑀸𑀤𑀺𑀢𑀸𑀦𑀺 ¶ ,

𑀢𑀽𑀭𑀺𑀬𑀸𑀦𑀺𑀘𑀅𑀫𑁆𑀩𑀭𑁂;

𑀫𑀥𑀼𑀭𑁂𑀦𑀧𑀦𑀸𑀤𑀺𑀁𑀲𑀼,

𑀬𑀲𑀁𑀲𑀭𑁂𑀦𑀢𑀸𑀯𑀤𑁂.

𑁧𑁦.

𑀣𑀼𑀢𑀺𑀕𑀻𑀢𑀸𑀦𑀺𑀕𑀸𑀬𑀦𑁆𑀢𑀸,

𑀤𑁂𑀯𑀸𑀘𑀓𑀫𑀮𑀸𑀲𑀦𑀸;

𑀲𑀫𑁆𑀫𑁄𑀤𑁂𑀦𑁆𑀢𑀸𑀯𑀅𑀝𑁆𑀞𑀁𑀲𑀼 ¶ ,

𑀩𑀼𑀤𑁆𑀥𑀗𑁆𑀓𑀼𑀭𑀲𑁆𑀲𑀲𑀦𑁆𑀢𑀺𑀓𑁂.

𑁧𑁧.

𑀓𑀼𑀫𑀸𑀭𑁄𑀘𑀸𑀭𑀼𑀥𑀸𑀢𑀻𑀳𑀺,

𑀢𑁄𑀲𑀺𑀢𑀸𑀭𑀸𑀚𑀫𑀦𑁆𑀤𑀺𑀭𑁂;

𑀯𑀼𑀤𑁆𑀥𑁄𑀲𑀺𑀲𑀼𑀲𑀼𑀔𑀁𑀔𑁂𑀫𑀁,

𑀲𑀓𑁆𑀬𑀓𑀜𑁆𑀜𑀸𑀦𑀫𑀦𑁆𑀢𑀭𑁂.

𑁧𑁨.

𑀲𑁄𑀴𑀲𑀯𑀲𑁆𑀲𑀺𑀓𑁄𑀲𑀓𑁆𑀬 ¶ ,

𑀦𑀦𑁆𑀤𑀦𑁄𑀧𑀺𑀢𑀭𑀸𑀲𑀢𑀸;

𑀭𑀸𑀚𑀢𑁆𑀢𑁂𑀅𑀪𑀺𑀲𑀺𑀢𑁆𑀢𑁄𑀲𑀺,

𑀫𑁄𑀤𑁂𑀲𑀺𑀢𑀼𑀯𑀦𑀢𑁆𑀢𑀬𑁂.

𑁧𑁩.

𑀦𑀦𑁆𑀤𑀦𑀸𑀦𑀁𑀲𑁄𑀯𑀡𑁆𑀡 ¶ ,

𑀯𑀡𑁆𑀡𑀲𑁄𑀪𑀁𑀬𑀲𑁄𑀥𑀭𑀁;

𑀲𑁄𑀓𑀸𑀭𑁂𑀲𑀺𑀫𑀳𑁂𑀲𑀺𑀢𑁆𑀢𑀁,

𑀮𑁄𑀓𑀦𑁂𑀢𑁆𑀢𑀭𑀲𑀸𑀬𑀦𑀁.

𑁧𑁪.

𑀏𑀓𑀽𑀦𑀢𑀺𑀁𑀲𑀫𑁂𑀯𑀲𑁆𑀲𑁂,

𑀯𑀺𑀲𑀸𑀔𑀧𑀼𑀡𑁆𑀡𑀫𑁂𑀯𑀭𑁂;

𑀲𑀳𑀲𑁂𑀦𑀸𑀬𑀉𑀬𑁆𑀬𑀸𑀲𑀺,

𑀬𑀸𑀦𑁂𑀦𑀼𑀬𑁆𑀬𑀸𑀦𑀫𑀡𑁆𑀟𑀮𑀁.

𑁧𑁫.

𑀦𑀺𑀫𑀺𑀢𑁆𑀢𑁂𑀘𑀢𑀼𑀭𑁄𑀤𑀺𑀲𑁆𑀯𑀸 ¶ ,

𑀧𑀼𑀢𑁆𑀢𑀲𑁆𑀲𑀘𑀯𑀺𑀚𑀸𑀬𑀦𑀁;

𑀲𑀼𑀢𑁆𑀯𑀸𑀉𑀩𑁆𑀩𑀺𑀕𑁆𑀕𑀘𑀺𑀢𑁆𑀢𑁂𑀦,

𑀦𑀭𑀫𑀺𑀢𑁆𑀣𑀧𑀼𑀭𑁂𑀢𑀳𑀺𑀁.

𑁧𑁬.

𑀧𑀘𑁆𑀘𑀸𑀕𑀫𑁆𑀫𑀧𑀼𑀭𑀁𑀢𑀫𑁆𑀳𑀸 ¶ ,

𑀧𑀢𑁆𑀢𑁄𑀧𑀸𑀲𑀸𑀤𑀫𑀡𑁆𑀟𑀮𑀁;

𑀦𑀺𑀧𑁆𑀧𑀚𑁆𑀚𑀺𑀤𑁂𑀯𑀭𑀸𑀚𑀸𑀯,

𑀲𑀬𑀦𑁂𑀲𑁄𑀫𑀳𑀸𑀭𑀳𑁂.

𑁧𑁭.

𑀲𑀼𑀦𑁆𑀤𑀭𑀻𑀢𑀁𑀧𑀼𑀭𑀓𑁆𑀔𑀺𑀢𑁆𑀯𑀸 ¶ ,

𑀦𑀘𑁆𑀘𑀕𑀻𑀢𑁂𑀦𑀢𑁄𑀲𑀬𑀼𑀁;

𑀏𑀯𑀁𑀧𑀺𑀲𑁄𑀦𑀭𑀫𑀺𑀢𑁆𑀯𑀸,

𑀆𑀳𑀼𑀦𑀺𑀓𑁆𑀔𑀫𑀫𑀸𑀦𑀲𑁄.

𑁧𑁮.

𑀧𑀣𑀫𑀁𑀭𑀸𑀳𑀼𑀮𑀁𑀧𑀼𑀢𑁆𑀢𑀁,

𑀧𑀲𑁆𑀲𑀺𑀢𑁆𑀯𑀸𑀦𑀺𑀓𑁆𑀔𑀫𑀸𑀫𑀳𑀁;

𑀇𑀢𑀺𑀘𑀺𑀦𑁆𑀢𑀺𑀬𑀫𑀸𑀦𑁂𑀲𑁄,

𑀅𑀕𑀸𑀚𑀸𑀬𑀸𑀦𑀺𑀯𑁂𑀲𑀦𑀁;

𑁧𑁯.

𑀣𑀧𑁂𑀢𑁆𑀯𑀸𑀧𑀸𑀤𑀤𑀼𑀫𑁆𑀫𑀸𑀭𑁂 ¶ ,

𑀕𑀻𑀯𑀫𑀦𑁆𑀢𑁄𑀧𑀯𑁂𑀲𑀬𑀁;

𑀑𑀮𑁄𑀓𑀬𑀁𑀲𑀧𑀲𑁆𑀲𑀺𑀢𑁆𑀣,

𑀦𑀺𑀧𑁆𑀧𑀦𑁆𑀦𑀁𑀢𑀁𑀲𑀫𑀸𑀢𑀭𑀁.

𑁨𑁦.

𑀕𑀡𑁆𑀳𑁂𑀬𑁆𑀬𑀁𑀢𑀁𑀅𑀧𑀦𑁂𑀢𑁆𑀯𑀸 ¶ ,

𑀩𑀸𑀳𑀼𑀁𑀘𑁂𑀤𑁂𑀯𑀺𑀬𑀸𑀅𑀳𑀁;

𑀦𑀤𑀤𑁂𑀦𑀺𑀓𑁆𑀔𑀫𑁄𑀓𑀸𑀲𑀁,

𑀧𑀩𑀼𑀚𑁆𑀛𑀺𑀢𑁆𑀯𑀸𑀬𑀲𑁄𑀥𑀭𑀸.

𑁨𑁧.

𑀩𑀼𑀤𑁆𑀥𑀧𑀢𑁆𑀢𑁄𑀯𑀧𑀲𑁆𑀲𑁂𑀬𑁆𑀬𑀁,

𑀭𑀸𑀳𑀼𑀮𑀁𑀇𑀢𑀺𑀘𑀺𑀦𑁆𑀢𑀺𑀬;

𑀓𑀡𑁆𑀟𑀓𑀁𑀆𑀭𑀼𑀳𑀺𑀢𑁆𑀯𑀸𑀦,

𑀢𑀫𑁆𑀳𑀸𑀙𑀦𑁆𑀦𑁂𑀦𑀦𑀺𑀓𑁆𑀔𑀫𑀺.

𑁨𑁨.

𑀤𑁆𑀯𑀸𑀭𑀸𑀲𑀬𑀸𑀯𑀺𑀯𑀭𑀺𑀁𑀲𑀼 ¶ ,

𑀤𑁂𑀯𑀢𑀸𑀤𑁆𑀯𑀸𑀭𑀫𑀡𑁆𑀟𑀮𑀁;

𑀭𑀢𑀦𑀼𑀓𑁆𑀓𑀸𑀲𑀳𑀲𑁆𑀲𑀸𑀦𑀺,

𑀚𑀸𑀮𑀬𑀺𑀁𑀲𑀼𑀫𑀭𑀽𑀢𑀳𑀺𑀁.

𑁨𑁩.

𑀫𑀸𑀦𑀺𑀓𑁆𑀔𑀫𑀲𑁆𑀲𑀼𑀲𑀺𑀤𑁆𑀥𑀢𑁆𑀣,

𑀇𑀢𑁄𑀢𑁆𑀯𑀁𑀲𑀢𑁆𑀢𑀫𑁂𑀤𑀺𑀦𑁂;

𑀪𑀯𑀺𑀲𑁆𑀲𑀲𑁂𑀘𑀓𑁆𑀓𑀯𑀢𑁆𑀢𑀻 ¶ ,

𑀇𑀢𑀺𑀦𑀻𑀯𑀸𑀭𑀺𑀧𑀸𑀧𑀺𑀫𑀸.

𑁨𑁪.

𑀢𑀕𑁆𑀖𑀚𑀸𑀦𑀸𑀫𑀺𑀢𑀁𑀫𑀸𑀭,

𑀇𑀥𑀫𑀸𑀢𑀺𑀝𑁆𑀞𑀧𑀓𑁆𑀓𑀫;

𑀇𑀢𑀺𑀫𑀸𑀭𑀁 𑀧𑀮𑀸𑀧𑁂𑀢𑁆𑀯𑀸,

𑀯𑀺𑀲𑁆𑀲𑀝𑁆𑀞𑁄𑀲𑁄𑀅𑀪𑀺𑀓𑁆𑀓𑀫𑀺.

𑁨𑁫.

𑀧𑀢𑁆𑀯𑀸𑀦𑁄𑀫𑀦𑀤𑀻𑀢𑀻𑀭𑀁 ¶ ,

𑀧𑀼𑀦𑀸𑀕𑀫𑁂𑀲𑀺𑀢𑀼𑀭𑀕𑀁;

𑀙𑀦𑁆𑀦𑀜𑁆𑀘𑀦𑀕𑀭𑀁 𑀩𑀺𑀫𑁆𑀩𑀸;

𑀤𑁂𑀯𑀺𑀮𑀸𑀧𑀝𑀺𑀯𑁂𑀤𑀺𑀢𑀼𑀁.

𑁨𑁬.

𑀩𑀦𑁆𑀥𑀺𑀢𑀁𑀬𑀲𑁄𑀥𑀭𑀸𑀬,

𑀫𑁄𑀴𑀺𑀁𑀲𑀼𑀕𑀦𑁆𑀥𑀯𑀸𑀲𑀺𑀢𑀁;

𑀙𑁂𑀢𑁆𑀯𑀸𑀔𑀕𑁆𑀕𑁂𑀦𑀢𑀺𑀡𑁆𑀳𑁂𑀦,

𑀉𑀓𑁆𑀔𑀺𑀧𑀺𑀢𑁆𑀣𑀢𑀫𑀫𑁆𑀩𑀭𑁂.

𑁨𑁭.

𑀅𑀝𑁆𑀞𑀲𑁂𑀝𑁆𑀞𑀧𑀭𑀺𑀓𑁆𑀔𑀸𑀭𑁂 ¶ ,

𑀥𑀸𑀭𑁂𑀢𑁆𑀯𑀸𑀩𑁆𑀭𑀳𑁆𑀫𑀼𑀦𑀸𑀪𑀢𑁂;

𑀳𑀺𑀢𑁆𑀯𑀸𑀭𑀸𑀚𑀸𑀭𑀳𑀁𑀤𑀼𑀲𑁆𑀲𑀁,

𑀲𑀫𑀡𑀢𑁆𑀣𑀫𑀼𑀧𑀸𑀕𑀫𑀺.

𑁨𑁮.

𑀓𑀭𑁄𑀦𑁆𑀢𑁄𑀤𑀼𑀓𑁆𑀓𑀭𑀁𑀓𑀫𑁆𑀫𑀁 ¶ ,

𑀙𑀩𑁆𑀩𑀲𑁆𑀲𑀸𑀦𑀺𑀧𑀭𑀓𑁆𑀓𑀫𑁄;

𑀑𑀴𑀸𑀭𑀺𑀓𑀦𑁆𑀦𑀧𑀸𑀦𑁂𑀳𑀺;

𑀤𑁂𑀳𑀁𑀬𑀸𑀧𑁂𑀲𑀺𑀞𑀻𑀢𑀺𑀬𑀸.

𑁨𑁯.

𑀯𑀺𑀲𑀸𑀔𑀧𑀼𑀡𑁆𑀡𑀫𑁂𑀳𑁂𑀫,

𑀧𑀸𑀢𑀺𑀁𑀧𑀸𑀬𑀲𑀧𑀽𑀭𑀺𑀢𑀁;

𑀲𑀼𑀚𑀸𑀢𑀸𑀬𑀸𑀪𑀢𑀁𑀕𑀬𑁆𑀳,

𑀦𑁂𑀭𑀜𑁆𑀘𑀭𑀫𑀼𑀧𑀸𑀕𑀫𑀺.

𑁩𑁦.

𑀦𑁂𑀭𑀜𑁆𑀘𑀭𑀸𑀬𑀢𑀻𑀭𑀫𑁆𑀳𑀺 ¶ ,

𑀲𑀼𑀢𑁆𑀯𑀸𑀧𑀸𑀬𑀸𑀲𑀫𑀼𑀢𑁆𑀢𑀫𑀁;

𑀧𑀝𑀺𑀬𑀢𑁆𑀣𑀯𑀭𑀫𑀕𑁆𑀕𑁂𑀦,

𑀩𑁄𑀥𑀺𑀫𑀡𑁆𑀟𑀫𑀼𑀧𑀸𑀕𑀫𑀺.

𑁩𑁧.

𑀚𑀬𑀸𑀲𑀦𑁂𑀧𑀮𑁆𑀮𑀗𑁆𑀓𑀫𑁆𑀳𑀺 ¶ ,

𑀞𑀺𑀢𑁄𑀥𑀺𑀝𑁆𑀞𑁂𑀲𑀺𑀥𑀻𑀢𑀺𑀫𑀸;

𑀦𑀯𑀻𑀭𑀺𑀬𑀁𑀚𑀳𑀺𑀲𑁆𑀲𑀸𑀫𑀺,

𑀅𑀧𑀢𑁆𑀢𑁄𑀇𑀢𑀺𑀩𑀼𑀤𑁆𑀥𑀢𑀁.

𑁩𑁨.

𑀤𑁂𑀯𑀺𑀦𑁆𑀤𑁄𑀥𑀫𑀬𑀁𑀲𑀗𑁆𑀔𑀁 ¶ ,

𑀲𑁂𑀢𑀘𑁆𑀙𑀢𑁆𑀢𑀁𑀧𑀺𑀢𑀸𑀫𑀳𑁄;

𑀥𑀸𑀭𑀬𑀁𑀯𑀸𑀤𑁂𑀦𑁆𑀢𑁄𑀯𑀻𑀡𑀁,

𑀧𑀜𑁆𑀘𑀲𑀻𑀔𑁄𑀞𑀺𑀢𑁄𑀢𑀳𑀺𑀁.

𑁩𑁩.

𑀣𑀼𑀢𑀺𑀕𑀻𑀢𑀸𑀦𑀺𑀕𑀸𑀬𑀦𑁆𑀢𑀻,

𑀅𑀝𑁆𑀞𑀸𑀲𑀼𑀁𑀤𑁂𑀯𑀲𑀼𑀦𑁆𑀤𑀭𑀻;

𑀕𑀳𑁂𑀢𑁆𑀯𑀸𑀳𑁂𑀫𑀫𑀜𑁆𑀚𑀽𑀲𑀸,

𑀲𑀼𑀭𑀧𑀼𑀧𑁆𑀨𑁂𑀳𑀺𑀧𑀽𑀭𑀺𑀢𑀸.

𑁩𑁪.

𑀅𑀦𑀢𑁆𑀣𑀫𑁂𑀦𑁆𑀢𑁂𑀲𑀽𑀭𑀺𑀬𑁂 ¶ ,

𑀫𑀸𑀭𑀲𑁂𑀦𑀁𑀧𑀮𑀸𑀧𑀬𑀻;

𑀅𑀭𑀼𑀡𑀼𑀕𑁆𑀕𑀫𑀦𑁂𑀓𑀸𑀮𑁂,

𑀧𑀢𑁆𑀢𑁄𑀲𑀺𑀩𑀼𑀤𑁆𑀥𑀢𑀁𑀯𑀭𑀁.

𑁩𑁫.

𑀧𑀝𑁆𑀞𑀸𑀦𑀁𑀲𑀫𑁆𑀫𑀲𑀦𑁆𑀢𑀲𑁆𑀲,

𑀦𑀸𑀣𑀲𑁆𑀲𑀤𑁂𑀳𑀢𑁄𑀲𑀼𑀪𑀸;

𑀙𑀩𑁆𑀩𑀡𑁆𑀡𑀪𑀸𑀦𑀺𑀘𑁆𑀙𑀥𑀦𑁆𑀢𑀺 ¶ ,

𑀯𑀺𑀥𑀸𑀯𑀦𑁆𑀢𑀺 𑀢𑀳𑀺𑀁 𑀢𑀳𑀺𑀁.

𑁩𑁬.

𑀦𑀻𑀮𑀸𑀦𑀻𑀮𑀸𑀧𑀺𑀢𑀸𑀧𑀻𑀢𑀸,

𑀭𑀢𑁆𑀢𑀸𑀭𑀢𑁆𑀢𑀸𑀘𑀦𑀺𑀘𑁆𑀙𑀭𑁂;

𑀲𑁂𑀢𑀸𑀲𑁂𑀢𑀸𑀘𑀫𑀜𑁆𑀘𑀺𑀝𑁆𑀞𑀸,

𑀢𑀫𑁆𑀳𑀸𑀢𑀫𑁆𑀳𑀸𑀧𑀪𑀲𑁆𑀲𑀭𑀸.

𑁩𑁭.

𑀩𑀼𑀤𑁆𑀥𑀧𑀢𑁆𑀢𑁄𑀧𑀺𑀩𑁄𑀥𑁂𑀬𑁆𑀬 ¶ ,

𑀥𑀫𑁆𑀫𑀕𑀫𑁆𑀪𑀻𑀭𑀢𑀁𑀚𑀺𑀦𑁄;

𑀦𑀺𑀲𑀸𑀫𑁂𑀢𑁆𑀯𑀸𑀦𑀤𑁂𑀲𑁂𑀢𑀼𑀁,

𑀦𑀺𑀭𑀼𑀲𑁆𑀲𑀸𑀳𑀫𑀼𑀧𑀸𑀕𑀢𑁄.

𑁩𑁮.

𑀲𑀳𑀫𑁆𑀧𑀢𑀺𑀢𑀤𑀸𑀩𑁆𑀭𑀳𑁆𑀫𑀸 ¶ ,

𑀤𑁂𑀲𑀦𑀸𑀬𑀸𑀪𑀺𑀬𑀸𑀘𑀢𑀺;

𑀢𑁂𑀦𑀲𑁄𑀧𑀣𑀫𑀁𑀲𑁂𑀝𑁆𑀞𑀁,

𑀥𑀫𑁆𑀫𑀘𑀓𑁆𑀓𑀫𑀤𑁂𑀲𑀬𑀺.

𑁩𑁯.

𑀏𑀓𑀯𑀲𑁆𑀲𑁄𑀯𑀲𑀫𑁆𑀩𑀼𑀤𑁆𑀥𑁄 ¶ ,

𑀯𑀭𑁂𑀨𑀕𑁆𑀕𑀼𑀡𑀧𑀼𑀡𑁆𑀡𑀫𑁂;

𑀲𑀓𑁆𑀬𑀸𑀦𑀁𑀦𑀕𑀭𑀁𑀕𑀜𑁆𑀙𑀻,

𑀲𑀸𑀯𑀓𑁂𑀳𑀺𑀧𑀼𑀭𑀓𑁆𑀔𑀢𑁄.

𑁪𑁦.

𑀲𑀫𑀸𑀕𑀫𑀫𑁆𑀳𑀺𑀜𑀸𑀢𑀻𑀦𑀁,

𑀯𑀲𑁆𑀲𑀸𑀧𑁂𑀲𑀺𑀲𑀼𑀲𑀻𑀢𑀮𑀁;

𑀫𑀳𑀸𑀫𑁂𑀖𑀜𑁆𑀘𑀤𑁂𑀲𑁂𑀲𑀺 ¶ ,

𑀯𑁂𑀲𑁆𑀲𑀦𑁆𑀢𑀭𑀲𑁆𑀲𑀚𑀸𑀢𑀓𑀁.

𑁪𑁧.

𑀕𑀦𑁆𑀢𑁆𑀯𑀸𑀲𑀸𑀯𑀓𑀬𑀼𑀕𑁂𑀦,

𑀬𑀲𑁄𑀥𑀭𑀸𑀬𑀫𑀦𑁆𑀤𑀺𑀭𑀁;

𑀢𑀁𑀢𑁄𑀲𑁂𑀢𑁆𑀯𑀸𑀦𑀤𑁂𑀲𑁂𑀲𑀺,

𑀘𑀦𑁆𑀤𑀓𑀺𑀦𑁆𑀦𑀭𑀚𑀸𑀢𑀓𑀁.

𑁪𑁨.

𑀯𑀲𑀺𑀢𑁆𑀯𑀸𑀲𑀢𑁆𑀢𑀫𑁂𑀯𑀲𑁆𑀲𑁂 ¶ ,

𑀢𑀸𑀯𑀢𑀺𑀁𑀲𑁂 𑀲𑀺𑀮𑀸𑀲𑀦𑁂;

𑀅𑀪𑀺𑀥𑀫𑁆𑀫𑁂𑀦 𑀢𑁄𑀲𑁂𑀲𑀺,

𑀤𑁂𑀯𑀩𑁆𑀭𑀳𑁆𑀫𑀕𑀡𑀁𑀩𑀳𑀼𑀁.

𑁪𑁩.

𑀧𑀢𑁆𑀣𑀭𑀁𑀜𑀸𑀡𑀚𑀸𑀮𑁂𑀦,

𑀮𑁄𑀓𑀁𑀮𑁄𑀓𑁂𑀲𑀺𑀲𑀩𑁆𑀩𑀤𑀸;

𑀩𑁄𑀥𑁂𑀢𑀼𑀁𑀩𑁄𑀥𑀺𑀬𑀸𑀪𑀸𑀬;

𑀩𑁄𑀥𑁂𑀬𑁆𑀬𑀚𑀦𑀧𑀗𑁆𑀓𑀚𑀁.

𑁪𑁪.

𑀩𑁄𑀥𑀦𑁂𑀬𑁆𑀬𑀁𑀚𑀦𑀁𑀤𑀺𑀲𑁆𑀯𑀸 ¶ ,

𑀘𑀓𑁆𑀓𑀯𑀸𑀴𑀧𑀭𑀁𑀧𑀭𑀸;

𑀕𑀦𑁆𑀢𑁆𑀯𑀸𑀧𑀺𑀯𑀺𑀩𑁄𑀥𑁂𑀲𑀺,

𑀩𑀼𑀤𑁆𑀥𑀲𑁂𑀝𑁆𑀞𑁄𑀲𑀼𑀩𑁄𑀥𑀺𑀫𑀸.

𑁪𑁫.

𑀩𑀓𑀸𑀤𑀺𑀓𑁂𑀘𑀩𑁆𑀭𑀳𑁆𑀫𑀸𑀦𑁄,

𑀬𑀓𑁆𑀔𑁂𑀘𑀸𑀴𑀯𑀓𑀸𑀤𑀺𑀓𑁂;

𑀧𑀼𑀭𑀺𑀦𑁆𑀤𑀤𑀸𑀤𑀺𑀓𑁂𑀤𑁂𑀯𑁂,

𑀲𑀽𑀘𑀺𑀮𑁄𑀫𑀸𑀤𑀺𑀭𑀓𑁆𑀔𑀲𑁂.

𑁪𑁬.

𑀚𑀸𑀡𑀼𑀲𑁄𑀡𑀸𑀤𑀺𑀓𑁂𑀦𑁂𑀓𑁂 ¶ ,

𑀩𑁆𑀭𑀸𑀳𑁆𑀫𑀡𑁂𑀘𑀸𑀯𑀺𑀓𑀓𑁆𑀔𑀴𑁂;

𑀘𑀽𑀴𑁄𑀤𑀭𑀸𑀤𑀺𑀓𑁂𑀦𑀸𑀕𑁂,

𑀥𑀦𑀧𑀸𑀮𑀸𑀤𑀺𑀓𑁂𑀕𑀚𑁂.

𑁪𑁭.

𑀩𑀼𑀤𑁆𑀥𑀸𑀫𑀬𑀁𑀢𑀺𑀫𑀜𑁆𑀜𑀦𑁆𑀢𑁂,

𑀢𑀺𑀢𑁆𑀣𑀺𑀬𑁂𑀦𑁂𑀓𑀮𑀤𑁆𑀥𑀺𑀓𑁂;

𑀤𑀫𑁂𑀲𑀺𑀤𑀫𑀢𑀁𑀲𑁂𑀝𑁆𑀞𑁄,

𑀧𑀭𑀯𑀸𑀤𑀧𑁆𑀧𑀫𑀤𑁆𑀤𑀦𑁄.

𑁪𑁮.

𑀅𑀦𑀼𑀩𑁆𑀬𑀜𑁆𑀚𑀦𑀲𑀫𑁆𑀧𑀦𑁆𑀦𑁄 ¶ ,

𑀏𑀲𑀩𑀸𑀢𑁆𑀢𑀺𑀁𑀲𑀮𑀓𑁆𑀔𑀡𑁄;

𑀩𑁆𑀬𑀸𑀫𑀧𑁆𑀧𑀪𑀸𑀧𑀭𑀺𑀓𑁆𑀔𑀺𑀢𑁆𑀢𑁄,

𑀓𑁂𑀢𑀼𑀫𑀸𑀮𑀸𑀲𑀺𑀮𑀗𑁆𑀔𑀢𑁄.

𑁪𑁯.

𑀘𑀢𑀼𑀭𑀸𑀲𑀻𑀢𑀺𑀲𑀳𑀲𑁆𑀲 ¶ ,

𑀥𑀫𑁆𑀫𑀓𑁆𑀔𑀦𑁆𑀥𑁂𑀲𑀤𑁂𑀲𑀬𑀺;

𑀧𑀜𑁆𑀘𑀢𑀸𑀣𑀻𑀲𑀯𑀲𑁆𑀲𑀸𑀦𑀺,

𑀢𑀺𑀮𑁄𑀓𑀲𑁆𑀲𑀳𑀺𑀢𑀸𑀬𑀯𑁂.

𑁫𑁦.

𑀩𑀼𑀤𑁆𑀥𑀓𑀺𑀘𑁆𑀘𑀁𑀓𑀭𑁄𑀦𑁆𑀢𑁄𑀲𑁄,

𑀯𑀳𑀁𑀮𑁄𑀓𑀳𑀺𑀢𑀁𑀯𑀭𑀁;

𑀫𑁄𑀘𑀬𑀁𑀩𑀦𑁆𑀥𑀦𑀸𑀲𑀢𑁆𑀢𑁂 ¶ ,

𑀓𑀸𑀮𑀁𑀔𑁂𑀧𑁂𑀢𑀺𑀲𑀩𑁆𑀩𑀤𑀸.

𑁫𑁧.

𑀬𑀳𑀺𑀁 𑀆𑀲𑀻𑀢𑀺𑀓𑁄𑀦𑀸𑀣𑁄,

𑀦𑀺𑀩𑁆𑀩𑀼𑀢𑁄𑀯𑀭𑀫𑀜𑁆𑀚𑀓𑁂;

𑀫𑀳𑀸𑀲𑀸𑀮𑀯𑀦𑁂𑀭𑀫𑁆𑀫𑁂,

𑀫𑀮𑁆𑀮𑀸𑀦𑀁𑀦𑀺𑀕𑀫𑁂𑀢𑀳𑀺𑀁.

𑁫𑁨.

𑀅𑀕𑁆𑀕𑀺𑀓𑁆𑀔𑀦𑁆𑀥𑁄𑀯𑀮𑁄𑀓𑀲𑁆𑀫𑀺𑀁 ¶ ,

𑀚𑀸𑀮𑁂𑀢𑁆𑀯𑀸𑀦𑀲𑀲𑀸𑀯𑀓𑁄;

𑀦𑀺𑀩𑁆𑀩𑀸𑀬𑀺𑀢𑁆𑀣𑀲𑀗𑁆𑀔𑀢𑀸𑀦𑀁,

𑀧𑀓𑀸𑀲𑁂𑀦𑁆𑀢𑁄𑀅𑀦𑀺𑀘𑁆𑀘𑀢𑀁𑀢𑀺.

‘‘𑀦𑀫𑁄𑀢𑁂𑀪𑀕𑀯𑀸𑀦𑀸𑀣 ¶ ,

𑀦𑀫𑁄𑀢𑁂𑀚𑀬𑀢𑀁𑀚𑀬;

𑀦𑀫𑁄𑀢𑁂𑀲𑀺𑀭𑀺𑀲𑀫𑁆𑀧𑀦𑁆𑀦,

𑀦𑀫𑁄𑀢𑁂𑀮𑁄𑀓𑀦𑀦𑁆𑀤𑀦.’’

𑀩𑀼𑀤𑁆𑀥𑀯𑀦𑁆𑀤𑀦𑀸

𑁧.

𑀤𑁂𑀯𑀮𑁄𑀓𑀸 𑀘𑀯𑀺𑀢𑁆𑀯𑀸𑀦 ¶ ,

𑀫𑀳𑀸𑀫𑀸𑀬𑀸𑀬 𑀓𑀼𑀘𑁆𑀙𑀺𑀬𑀁;

𑀉𑀧𑁆𑀧𑀚𑁆𑀚𑀺 𑀕𑀼𑀭𑀼𑀯𑀸𑀭𑀫𑁆𑀳𑀺,

𑀯𑀦𑁆𑀤𑁂𑀢𑀁 𑀲𑀓𑁆𑀬𑀧𑀼𑀗𑁆𑀕𑀯𑀁.

𑁨.

𑀤𑀲𑀫𑀸 𑀲𑀘𑁆𑀘𑀬𑁂 𑀦𑁂𑀲𑁄 ¶ ,

𑀯𑀺𑀚𑀸𑀬𑀺 𑀫𑀸𑀢𑀼𑀓𑀼𑀘𑁆𑀙𑀺𑀢𑁄;

𑀲𑀼𑀓𑁆𑀓𑀯𑀸𑀭𑁂 𑀮𑀼𑀫𑁆𑀩𑀺𑀦𑀺𑀬𑀁,

𑀯𑀦𑁆𑀤𑁂 𑀢𑀁 𑀮𑁄𑀓𑀧𑀽𑀚𑀺𑀢𑀁.

𑁩.

𑀘𑀓𑁆𑀓𑀯𑀢𑁆𑀢𑀺𑀲𑀺𑀭𑀺𑀁 𑀳𑀺𑀢𑁆𑀯𑀸,

𑀫𑀳𑀸𑀲𑀺𑀦𑀺𑀓𑁆𑀔𑀫𑀁 𑀲𑀼𑀥𑀻;

𑀦𑀺𑀓𑁆𑀔𑀫𑀻 𑀘𑀦𑁆𑀤𑀯𑀸𑀭𑀫𑁆𑀳𑀺,

𑀦𑀫𑁂 𑀢𑀁 𑀫𑀼𑀦𑀺𑀓𑀼𑀜𑁆𑀚𑀭𑀁.

𑁪.

𑀅𑀲𑁆𑀲𑀢𑁆𑀣𑀩𑁄𑀥𑀺𑀫𑀽𑀮𑀫𑁆𑀳𑀺 ¶ ,

𑀧𑀮𑁆𑀮𑀗𑁆𑀓𑁂 𑀅𑀧𑀭𑀸𑀚𑀺𑀢𑁂;

𑀧𑀢𑁆𑀢𑁄 𑀲𑀩𑁆𑀩𑀜𑁆𑀜𑀼𑀢𑀁 𑀦𑀸𑀣𑁄,

𑀦𑀫𑁂 𑀢𑀁 𑀩𑀼𑀥𑀯𑀸𑀲𑀭𑁂.

𑁫.

𑀧𑀜𑁆𑀘𑀯𑀕𑁆𑀕𑀺𑀬𑀫𑀼𑀔𑀸𑀦𑀁 ¶ ,

𑀤𑁂𑀯𑀸𑀦𑀁 𑀫𑀺𑀕𑀤𑀸𑀯𑀦𑁂;

𑀲𑁄𑀭𑀺𑀯𑀸𑀭𑁂 𑀥𑀫𑁆𑀫𑀘𑀓𑁆𑀓𑀁,

𑀯𑀢𑁆𑀢𑁂𑀲𑀺 𑀢𑀁 𑀚𑀺𑀦𑀁 𑀦𑀫𑁂.

𑁬.

𑀲𑀩𑁆𑀩𑀲𑀗𑁆𑀔𑀢𑀥𑀫𑁆𑀫𑀸𑀦𑀁,

𑀧𑀓𑀸𑀲𑀺𑀬𑀅𑀦𑀺𑀘𑁆𑀘𑀢𑀁;

𑀦𑀺𑀩𑁆𑀩𑀼𑀢𑀁𑀅𑀗𑁆𑀕𑀯𑀸𑀭𑀫𑁆𑀳𑀺,

𑀦𑀫𑀸𑀥𑀺𑀅𑀫𑀢𑀦𑁆𑀤𑀤𑀁.

𑁭.

𑀲𑀽𑀭𑀺𑀬𑀯𑀸𑀭𑁂 𑀲𑁄𑀡𑁆𑀡𑀸𑀬 ¶ ,

𑀤𑁄𑀡𑀺𑀬𑀸 𑀚𑀺𑀦𑀯𑀺𑀕𑁆𑀕𑀳𑁄;

𑀅𑀕𑁆𑀕𑀺𑀦𑀼𑀚𑁆𑀚𑀮𑀺𑀢𑁄 𑀢𑀲𑁆𑀲,

𑀦𑀫𑁂 𑀥𑀸𑀢𑀼𑀲𑀭𑀻𑀭𑀓𑀁.

𑀉𑀡𑁆𑀡𑀸𑀮𑁄𑀫𑀺𑀓𑀦𑀸𑀣 𑀯𑀦𑁆𑀤𑀦𑀸

𑀉𑀡𑁆𑀡𑀸𑀮𑁄𑀥𑀺𑀓𑀦𑀸𑀣𑀲𑁆𑀲 ¶ ,

𑀉𑀡𑁆𑀡𑀸𑀬 𑀪𑀫𑀼𑀓𑀦𑁆𑀢𑀭𑁂;

𑀯𑀚𑀺𑀭𑀸𑀯𑀺𑀬 𑀲𑁄𑀪𑀦𑁆𑀢𑀺,

𑀦𑀺𑀓𑁆𑀔𑀦𑁆𑀢𑀺 𑀬𑁄 𑀲𑀼𑀧𑀡𑁆𑀟𑀭𑀸.

𑀯𑀚𑀺𑀭𑀕𑁆𑀖𑀦𑀓𑀸𑀬𑀲𑁆𑀲 ¶ ,

𑀦𑀸𑀣𑀲𑁆𑀲 𑀤𑁂𑀳𑀢𑁄 𑀲𑀼𑀪𑀸;

𑀯𑀚𑀺𑀭𑁂𑀬𑁆𑀬𑀸 𑀦𑀺𑀘𑁆𑀙𑀭𑀦𑁆𑀢𑀺;

𑀯𑀺𑀚𑁆𑀚𑀽𑀯 𑀕𑀕𑀡𑀦𑁆𑀢𑀭𑁂.

𑀭𑀺𑀦𑀺𑀦𑁆𑀤𑀫𑀦𑁆𑀤𑀳𑀸𑀲𑁂𑀦,

𑀤𑀸𑀞𑀸𑀬 𑀦𑀺𑀘𑁆𑀙𑀭𑀸 𑀧𑀪𑀸;

𑀯𑀺𑀧𑁆𑀨𑀽𑀭𑀺𑀲𑀸 𑀤𑀺𑀲𑀸𑀲𑀩𑁆𑀩𑀸,

𑀯𑀚𑀺𑀭𑀸𑀬𑀦𑁆𑀢𑀺 𑀧𑀡𑁆𑀟𑀭𑀸.
